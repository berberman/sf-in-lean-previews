5+--  # ProofObjects: The Curry-Howard Correspondence
6+
7+--  This setting is enabled to make some tricky mistakes
8+--  harder to make.
9+
10+set_option autoImplicit false
11+
12+--  "Algorithms are the computational content of proofs."
13+--  (Robert Harper)
14+
15+--  Programming and proving in Lean are two sides of the
16+--  same coin. Proofs manipulate evidence much as programs
17+--  manipulate data.
18+--
19+--  Question: If evidence is data, what are propositions
20+--  themselves?
21+--
22+--  Answer: They are types!
23+
24+--  Look again at the formal definition of the `Even`
25+--  property. For illustrative purposes, we've made the
26+--  implicit argument `{n : Nat}` to `Even.succ_succ`
27+--  explicit.
28+
29+namespace EvenProp
30+inductive Even : Nat → Prop where
31+  | zero : Even 0
32+  | succ_succ (n : Nat) (h : Even n) : Even (n + 2)
33+
34+--  We can pronounce the ":" here as either "has type" or
35+--  "is a proof of." For example, the second line in the
36+--  definition of `Even` declares that
37+--  `(Even.zero : Even 0)`. Instead of "`Even.zero` has type
38+--  `Even 0`," we can say that "`Even.zero` is a proof of
39+--  `Even 0`."
40+
41+--  This pun between types and propositions — between `:` as
42+--  "has type" and `:` as "is a proof of" or "is evidence
43+--  for" — is called the *Curry-Howard correspondence*. It
44+--  proposes a deep connection between the world of logic
45+--  and the world of computation:
46+--
47+--      propositions  ~  types
48+--      proofs        ~  programs
49+--
50+--  See Wadler (2015) for a brief history and modern
51+--  exposition.
52+
53+--  Many useful insights follow from this connection. To
54+--  begin with, it gives us a natural interpretation of the
55+--  type of the `Even.succ_succ` constructor.
56+--
57+--  The type of `Even.succ_succ` says that it is a
58+--  *function* (better: a data *constructor*) taking two
59+--  arguments (one number `n` plus evidence for `Even n`)
60+--  and returning evidence that `n + 2` is even.
61+
62+#check @Even.succ_succ
63+
64+--  Output:
65+--    Even.succ_succ : ∀ (n : Nat), Even n → Even (n + 2)
66+
67+--  This can be read "`Even.succ_succ` is a constructor that
68+--  takes two arguments — a number `n` and evidence for the
69+--  proposition `Even n` — and yields evidence for the
70+--  proposition `Even (n + 2)`."
71+
72+--  Now let's look again at an earlier proof involving
73+--  `Even`.
74+
75+theorem even_4 : Even 4 := by
76+  apply Even.succ_succ
77+  apply Even.succ_succ
78+  apply Even.zero
79+
80+--  Just as with ordinary data values and functions, we can
81+--  use the `#print` command to see the *proof object* that
82+--  results from this proof script.
83+
84+#print even_4
85+
86+--  Output:
87+--    theorem EvenProp.even_4 : Even 4 :=
88+--    Even.succ_succ 2 (Even.succ_succ 0 Even.zero)
89+
90+--  Indeed, we can also write down this proof object
91+--  directly, with no need for a proof script at all:
92+
93+#check (Even.succ_succ 2 (Even.succ_succ 0 Even.zero)
94+  : Even (2 + 2))
95+
96+--  Output:
97+--    Even.succ_succ 2 (Even.succ_succ 0 Even.zero) : Even (2 + 2)
98+
99+--  Similarly, as we've seen, we can directly apply theorems
100+--  to arguments in proof scripts:
101+
102+theorem even_4' : Even 4 := by
103+  apply Even.succ_succ 2 (Even.succ_succ 0 Even.zero)

105+--  ## Proof Scripts
106+
107+--  When we write a proof using tactics, what we are doing
108+--  is instructing Lean to build a proof object under the
109+--  hood. We can see this using the custom command
110+--  `show_proof`:
111+
112+--  THE FOLLOWING DETAILS CAN BE SKIPPED (Tactic: show_proof)
113+section ShowProof
114+open Lean Elab Tactic Meta
115+
116+/-- Like `instantiateMVars`, but also substitutes holes
117+whose assignment Lean delays until they are complete
118+(e.g., the body under an `intro`). -/
119+partial def instantiatePartial (e : Expr) : MetaM Expr := do
120+  let e ← instantiateMVars e
121+  transform e (pre := fun e => do
122+    let .mvar m := e.getAppFn | return .continue
123+    let some d ← getDelayedMVarAssignment? m
124+      | return .continue
125+    let v ← instantiatePartial (mkMVar d.mvarIdPending)
126+    let args := e.getAppArgs
127+    let n := d.fvars.size
128+    let v :=
129+      (v.abstract d.fvars).instantiateRevRange 0 n args
130+    return .done (mkAppN v args[n:]))
131+
132+/-- `show_proof tac₁ … tacₙ` runs the tactics in order,
133+printing the partial proof term (with `?_` for unfilled
134+holes) before the first tactic and after each one. -/
135+elab "show_proof" seq:tacticSeq : tactic => do
136+  let root ← getMainGoal
137+  let s := seq.raw[0]
138+  let tacs :=
139+    if s.getKind == ``Parser.Tactic.tacticSeqBracketed then
140+      s[1].getSepArgs
141+    else s[0].getSepArgs
142+  let render : TacticM Format := do
143+    let e ← instantiatePartial (mkMVar root)
144+    withOptions
145+      (fun o => (o.setBool `pp.mvars false).setBool
146+        `pp.fieldNotation false)
147+      (ppExpr e)
148+  let mut out ← render
149+  for t in tacs do
150+    evalTactic t
151+    let tac ← PrettyPrinter.ppTactic ⟨t⟩
152+    out := out ++ .line ++ .line ++ f!"after {tac}:" ++
153+      .nest 2 (.line ++ (← render))
154+  logInfo m!"{out}"
155+
156+end ShowProof
157+--  END DETAILS
158+
159+theorem even_4'' : Even 4 := by
160+  show_proof
161+    apply Even.succ_succ
162+    apply Even.succ_succ
163+    apply Even.zero
164+
165+--  Output:
166+--    ?_
167+--
168+--    after apply Even.succ_succ:
169+--      Even.succ_succ 2 ?_
170+--
171+--    after apply Even.succ_succ:
172+--      Even.succ_succ 2 (Even.succ_succ 0 ?_)
173+--
174+--    after apply Even.zero:
175+--      Even.succ_succ 2 (Even.succ_succ 0 Even.zero)

177+--  Tactic proofs are convenient, but they are not essential
178+--  in Lean: in principle, we can always just construct the
179+--  required evidence by hand. Then we can write a proof
180+--  term directly following a `:=` (instead of writing `by`
181+--  and then tactics) to introduce a global name for this
182+--  evidence.
183+
184+theorem even_4''' : Even 4 :=
185+  Even.succ_succ 2 (Even.succ_succ 0 Even.zero)

187+--  ## Quantifiers, Implications, Functions
188+
189+--  In Lean's computational universe (where data structures
190+--  and programs live), there are two sorts of values that
191+--  have arrows in their types: *constructors* introduced by
192+--  `inductive`ly defined data types, and *functions*.
193+--
194+--  Similarly, in Lean's logical universe (where we carry
195+--  out proofs), there are two ways of giving evidence for
196+--  an implication: constructors introduced by `inductive`ly
197+--  defined propositions, and... functions!
198+
199+--  For example, consider this statement:
200+
201+theorem even_plus4 (n : Nat) (h : Even n) :
202+    Even (n + 4) := by
203+  apply Even.succ_succ
204+  apply Even.succ_succ
205+  apply h
206+
207+--  What is the proof object corresponding to `even_plus4`?
208+
209+--  We're looking for an expression whose *type* is
210+--  `∀ (n : Nat), Even n → Even (n + 4)` — that is, a
211+--  *function* that takes two arguments (one number and a
212+--  piece of evidence) and returns a piece of evidence!
213+--
214+--  Here it is:
215+
216+theorem even_plus4' : ∀ (n : Nat), Even n → Even (n + 4) :=
217+  fun (n : Nat) => fun (h : Even n) =>
218+    Even.succ_succ (n + 2) (Even.succ_succ n h)
219+
220+--  Or equivalently:
221+
222+theorem even_plus4'' (n : Nat) (h : Even n) :
223+    Even (n + 4) :=
224+  Even.succ_succ (n + 2) (Even.succ_succ n h)
225+
226+#check (even_plus4'' : ∀ (n : Nat), Even n → Even (n + 4))
227+
228+--  ### Dependency, Equivalence, and Proof Irrelevance
229+
230+--  When we view the proposition being proved by
231+--  `even_plus4` as a function type, one interesting point
232+--  becomes apparent: The second argument's type, `Even n`,
233+--  mentions the *value* of the first argument, `n`.
234+--
235+--  While such *dependent types* are not found in most
236+--  mainstream programming languages, they can be quite
237+--  useful in programming too, as the flurry of activity in
238+--  the functional programming community over the past
239+--  couple of decades (and recently, most notably Lean!)
240+--  demonstrates. Dependent types sit at the "highest
241+--  corner" of the Lambda Cube (see the `Stlc` chapter of
242+--  *Type Systems*): types can depend on *values* in
243+--  addition to other types.
244+
245+--  Notice that both implication (`→`) and quantification
246+--  (`∀`) correspond to functions on evidence. In fact, they
247+--  are really the same thing: `→` is just a shorthand for a
248+--  degenerate use of `∀` where there is no dependency,
249+--  i.e., no need to give a name to the type on the
250+--  left-hand side of the arrow:
251+--
252+--         ∀ (x : Nat), Nat
253+--      =  ∀ (_ : Nat), Nat
254+--      =  Nat        → Nat
255+
256+--   ----------------------------------------
257+

259+

261+--

263+--        | zero : Even 0
264+--        | succ_succ (n : Nat) (h : Even n) : Even (n + 2)
265+--
266+--  What is the type of this expression?
267+--
268+--      fun (n : Nat) =>
269+--        fun (h : Even n) =>
270+--           Even.succ_succ (n + 2) (Even.succ_succ n h)
271+--
272+--  (A) `∀ (n : Nat), Even n`
273+--

275+--
276+--  (C) `∀ (n : Nat), Even n → Even n`
277+--
278+--  (D) `∀ (n : Nat), Even n → Even (n + 2)`
279+--

291+--  ## Practical Uses of Terms in Proofs
292+
293+--  In the `Even` examples, we show how to build proofs
294+--  using terms to illustrate that they are one and they
295+--  same. But in this particular example, doing so is
296+--  impractical: the proof terms above are verbose and
297+--  arguably harder to write and follow than a tactic proof,
298+--  which is more aligned with most people's style of
299+--  reasoning. So the question naturally arises- would we
300+--  ever use terms in *real* Lean proofs? The answer is yes,
301+--  and we show some examples where such use is both helpful
302+--  and actively encouraged in real Lean developments.
303+--
304+--  Here are some of the places where terms most often
305+--  appear in Lean proofs:
306+--  1. `rfl`: The *term* `rfl` is used very often as a full
307+--     proof instead of `by rfl` (which references the
308+--     *tactic* `rfl` of the same name) and is even
309+--     preferred in Lean's Mathlib, which writes its
310+--     `@[simp]` definitional lemmas this way.
311+
312+theorem two_add_two : 2 + 2 = 4 := rfl
313+
314+@[simp]
315+theorem length_cons {α : Type} (x : α) (xs : List α) :
316+    (x :: xs).length = xs.length + 1 := rfl
317+
318+--  2. To succinctly apply an existing lemma, possibly via
319+--     dot notation on proofs
320+
321+theorem add_comm_flip (a b : Nat) : b + a = a + b := Nat.add_comm b a
322+theorem le_chain {a b c : Nat} (h₁ : a ≤ b) (h₂ : b ≤ c) : a ≤ c :=
323+  Nat.le_trans h₁ h₂
324+
325+--  3. Within anonymous constructors and projections for
326+--     `And` (`∧`), `Iff` (`↔`), and `Exists` (`∃`)
327+
328+theorem and_swap {P Q : Prop} (h : P ∧ Q) : Q ∧ P := ⟨h.2, h.1⟩
329+theorem iff_swap {P Q : Prop} (h : P ↔ Q) : Q ↔ P := ⟨h.2, h.1⟩
330+theorem exists_double : ∃ n, n + n = 4 := ⟨2, rfl⟩
331+
332+--  4. `fun` for implications (as in `even_plus4'`)
333+
334+theorem imp_trans {P Q R : Prop} (hpq : P → Q) (hqr : Q → R) :
