1235+--  Here is an alternative "mathematical" definition of the factorial
1236+--  function:
1237+
1238+def realFact (n : Nat) : Nat :=
1239+  match n with
1240+  | 0 => 1
1241+  | n' + 1 => (n' + 1) * realFact n'

1243+--  We would like to show that they agree -- if we start `factCom` in a
1244+--  state where variable `X` contains some number `n`, then it will
1245+--  terminate in a state where variable `Y` contains the factorial of `n`.
1246+--
1247+--  To show this, we rely on the critical idea of a *loop invariant*.
1248+
1249+def FactInvariant (n : Nat) (st : State) : Prop :=
1250+  st[Y] * realFact st[Z] = realFact n
1251+
1252+--  We show that the body of the factorial loop preserves the invariant:
1253+
1254+theorem factBody_preserves_invariant {st st' : State} {n : Nat}
1255+    (hinv : FactInvariant n st) (hz : st[Z] ≠ 0)
1256+    (heval : st =[ ~factBody ]=> st') :
1257+    FactInvariant n st' := by
1258+  rw [FactInvariant] at hinv ⊢
1259+  rw [factBody] at heval
1260+  inversion heval with
1261+  | seq _ h₁ h₂ =>
1262+    inversion h₁ with
1263+    | asgn hy =>
1264+      inversion h₂ with
1265+      | asgn hz' =>
1266+        subst hy hz'
1267+        have hyz : Y ≠ Z := by decide

1269+        simp [hyz, hzy]
1270+        -- Show that `st[Z] = z + 1` for some `z`
1271+        cases hzz : st[Z] with
1272+        | zero => contradiction
1273+        | succ z =>

1275+          rw [Nat.add_sub_cancel, Nat.mul_assoc]
1276+          exact hinv

1279+--  invariant:

1281+theorem factLoop_preserves_invariant {st st' : State} {n : Nat}
1282+    (hinv : FactInvariant n st) (heval : st =[ ~factLoop ]=> st') :
