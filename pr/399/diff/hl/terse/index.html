973+--  Here is an alternative "mathematical" definition of the
974+--  factorial function:
975+
976+def realFact (n : Nat) : Nat :=

978+  | 0 => 1
979+  | n' + 1 => (n' + 1) * realFact n'

981+--  We would like to show that they agree -- if we start
982+--  `factCom` in a state where variable `X` contains some
983+--  number `n`, then it will terminate in a state where
984+--  variable `Y` contains the factorial of `n`.
985+--
986+--  To show this, we rely on the critical idea of a *loop
987+--  invariant*.

989+def FactInvariant (n : Nat) (st : State) : Prop :=
990+  st[Y] * realFact st[Z] = realFact n

992+--  We show that the body of the factorial loop preserves
993+--  the invariant:

995+theorem factBody_preserves_invariant {st st' : State} {n : Nat}
996+    (hinv : FactInvariant n st) (hz : st[Z] ≠ 0)
997+    (heval : st =[ ~factBody ]=> st') :
998+    FactInvariant n st' := by
999+  rw [FactInvariant] at hinv ⊢
1000+  rw [factBody] at heval
1001+  inversion heval with
1002+  | seq _ h₁ h₂ =>
1003+    inversion h₁ with
1004+    | asgn hy =>
1005+      inversion h₂ with
1006+      | asgn hz' =>
1007+        subst hy hz'
1008+        have hyz : Y ≠ Z := by decide

1010+        simp [hyz, hzy]
1011+        -- Show that `st[Z] = z + 1` for some `z`
1012+        cases hzz : st[Z] with
1013+        | zero => contradiction
1014+        | succ z =>
1015+          rw [hzz, realFact] at hinv
1016+          rw [Nat.add_sub_cancel, Nat.mul_assoc]
1017+          exact hinv

1565+--  raised (`s = sThrow`) or that execution can continue
1566+--  normally (`s = sNormal`).

1568+--    change and execution of any enclosing loop can
1569+--    continue normally.
1570+--  - If the command is `throw`, the state stays unchanged
1571+--    but we signal a `sThrow`.
1572+--  - If the command is `try`, we begin by executing the
1573+--    first subcommand; if it terminates normally, then the
1574+--    whole `try` terminates in the same state; if it
1575+--    terminates with an exception, we execute the second
1576+--    subcommand instead.
1577+--  - If the command is an assignment, then we update the
1578+--    binding for that variable in the state accordingly and
1579+--    signal that execution can continue normally.
1580+--  - If the command is of the form `if (b) {c₁} else {c₂}`,
1581+--    then the state is updated as in the original semantics

1583+--    the execution of whichever branch was taken.

1585+--    execute `c₁`. If this yields a `sThrow`, we skip the
1586+--    execution of `c₂` and propagate the `sThrow` signal to
1587+--    the surrounding context; the resulting state is the
1588+--    same as the one obtained by executing `c₁` alone.
