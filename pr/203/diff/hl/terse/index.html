22+-- ### Definitions

24+def Aexp.equiv (a₁ a₂ : Aexp) : Prop :=
25+  ∀ (st: State),
26+    a₁.eval st = a₂.eval st
27+
28+def Bexp.equiv (b₁ b₂ : Bexp) : Prop :=
29+  ∀ (st: State),
30+    b₁.eval st = b₂.eval st
31+
32+example : Aexp.equiv
33+          (aexp { X - X })
34+          (aexp { 0 }) :=
35+  by
36+    intros st
37+    simp
38+
39+example : Bexp.equiv
40+          (bexp { X - X = 0 })
41+          (bexp { true }) :=
42+  by
43+    intros st
44+    simp
45+
46+def Com.equiv (c₁ c₂: Com) : Prop :=
47+    ∀ (st st': State),
48+      (st =[ c₁ ]=> st') ↔ (st =[ c₂ ]=> st')
