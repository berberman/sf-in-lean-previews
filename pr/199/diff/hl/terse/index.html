864+-- _Details:_ Notation encoding: printing assertions back

866+namespace Assertion.Delab
867+open Lean PrettyPrinter Delaborator SubExpr Imp.Delab

869+/-- Rebuild the surface form of an assertion body, undoing the state
870+threading the `assn` elaborator performs: `st[X]` prints as `X`,
871+`Aexp.eval st a` as `a`, `Bexp.eval st b = true` as `b`, an applied
872+assertion `P st` as `P`, and a subterm that does not mention the state
873+prints as itself. -/
874+partial def delabBody (stId : FVarId) : DelabM Term := do
875+  let e ← getExpr
876+  if !e.containsFVar stId then
877+    delab
878+  else
879+    match_expr e with
880+    | MyGetElem.getElem _ _ _ _ st _ =>
881+      guard (st == .fvar stId)
882+      withNaryArg 5 delab
883+    | Aexp.eval st _ =>
884+      guard (st == .fvar stId)
885+      withAppArg delab
886+    | HAdd.hAdd _ _ _ _ _ _ =>
887+      `($(← withNaryArg 4 (delabBody stId)) + $(← withNaryArg 5 (delabBody stId)))
888+    | HSub.hSub _ _ _ _ _ _ =>
889+      `($(← withNaryArg 4 (delabBody stId)) - $(← withNaryArg 5 (delabBody stId)))
890+    | HMul.hMul _ _ _ _ _ _ =>
891+      `($(← withNaryArg 4 (delabBody stId)) * $(← withNaryArg 5 (delabBody stId)))
892+    | Eq _ l r =>
893+      -- `Bexp.eval st b = true` is the threaded form of a bare boolean `b`
894+      if r.isConstOf ``Bool.true && l.isAppOfArity ``Bexp.eval 2
895+          && l.appFn!.appArg! == .fvar stId then
896+        withNaryArg 1 <| withAppArg delab
897+      else
898+        `($(← withNaryArg 1 (delabBody stId)) = $(← withNaryArg 2 (delabBody stId)))
899+    | Ne _ _ _ =>
900+      `($(← withNaryArg 1 (delabBody stId)) ≠ $(← withNaryArg 2 (delabBody stId)))
901+    | LE.le _ _ _ _ =>
902+      `($(← withNaryArg 2 (delabBody stId)) ≤ $(← withNaryArg 3 (delabBody stId)))
903+    | LT.lt _ _ _ _ =>
904+      `($(← withNaryArg 2 (delabBody stId)) < $(← withNaryArg 3 (delabBody stId)))
905+    | GE.ge _ _ _ _ =>
906+      `($(← withNaryArg 2 (delabBody stId)) ≥ $(← withNaryArg 3 (delabBody stId)))
907+    | GT.gt _ _ _ _ =>
908+      `($(← withNaryArg 2 (delabBody stId)) > $(← withNaryArg 3 (delabBody stId)))
909+    | And _ _ =>
910+      `($(← withNaryArg 0 (delabBody stId)) ∧ $(← withNaryArg 1 (delabBody stId)))
911+    | Or _ _ =>
912+      `($(← withNaryArg 0 (delabBody stId)) ∨ $(← withNaryArg 1 (delabBody stId)))
913+    | Iff _ _ =>
914+      `($(← withNaryArg 0 (delabBody stId)) ↔ $(← withNaryArg 1 (delabBody stId)))
915+    | Not _ =>
916+      `(¬ $(← withAppArg (delabBody stId)))
917+    | _ =>
918+      if e.isArrow then
919+        `($(← withBindingDomain (delabBody stId)) →
920+          $(← withBindingBody `h (delabBody stId)))
921+      else if let .app f v := e then
922+        -- an applied assertion `P st` (or an applied escape lambda)
923+        guard (v == .fvar stId)
924+        guard !(f.containsFVar stId)
925+        withAppFn delab
926+      else
927+        failure

929+/-- Print an `Assertion`-valued term as it appears inside `{{ … }}`: a
930+state lambda is un-threaded; a term the printer cannot rebuild falls back
931+to the raw lambda, which is exactly this notation's escape form. -/
932+partial def delabAssn : DelabM Term := do
933+  if (← getExpr).isLambda then
934+    (withBindingBody' `st (pure ·.fvarId!) fun stId => delabBody stId)
935+      <|> Delaborator.delab
936+  else
937+    delab
938+

940+def delabTriple : Delab := whenPPOption getPPNotation do

950+def delabAssnArg (i : Nat) : DelabM Term := do
