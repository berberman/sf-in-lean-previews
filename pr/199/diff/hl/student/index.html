1040+/-- Rebuild the surface form of an assertion body, undoing the state
1041+threading the `assn` elaborator performs: `st[X]` prints as `X`,
1042+`Aexp.eval st a` as `a`, `Bexp.eval st b = true` as `b`, an applied
1043+assertion `P st` as `P`, and a subterm that does not mention the state
1044+prints as itself. -/
1045+partial def delabBody (stId : FVarId) : DelabM Term := do
1046+  let e ← getExpr
1047+  if !e.containsFVar stId then
1048+    delab
1049+  else
1050+    match_expr e with
1051+    | MyGetElem.getElem _ _ _ _ st _ =>
1052+      guard (st == .fvar stId)
1053+      withNaryArg 5 delab
1054+    | Aexp.eval st _ =>
1055+      guard (st == .fvar stId)
1056+      withAppArg delab
1057+    | HAdd.hAdd _ _ _ _ _ _ =>
1058+      `($(← withNaryArg 4 (delabBody stId)) + $(← withNaryArg 5 (delabBody stId)))
1059+    | HSub.hSub _ _ _ _ _ _ =>
1060+      `($(← withNaryArg 4 (delabBody stId)) - $(← withNaryArg 5 (delabBody stId)))
1061+    | HMul.hMul _ _ _ _ _ _ =>
1062+      `($(← withNaryArg 4 (delabBody stId)) * $(← withNaryArg 5 (delabBody stId)))
1063+    | Eq _ l r =>
1064+      -- `Bexp.eval st b = true` is the threaded form of a bare boolean `b`
1065+      if r.isConstOf ``Bool.true && l.isAppOfArity ``Bexp.eval 2
1066+          && l.appFn!.appArg! == .fvar stId then
1067+        withNaryArg 1 <| withAppArg delab
1068+      else
1069+        `($(← withNaryArg 1 (delabBody stId)) = $(← withNaryArg 2 (delabBody stId)))
1070+    | Ne _ _ _ =>
1071+      `($(← withNaryArg 1 (delabBody stId)) ≠ $(← withNaryArg 2 (delabBody stId)))
1072+    | LE.le _ _ _ _ =>
1073+      `($(← withNaryArg 2 (delabBody stId)) ≤ $(← withNaryArg 3 (delabBody stId)))
1074+    | LT.lt _ _ _ _ =>
1075+      `($(← withNaryArg 2 (delabBody stId)) < $(← withNaryArg 3 (delabBody stId)))
1076+    | GE.ge _ _ _ _ =>
1077+      `($(← withNaryArg 2 (delabBody stId)) ≥ $(← withNaryArg 3 (delabBody stId)))
1078+    | GT.gt _ _ _ _ =>
1079+      `($(← withNaryArg 2 (delabBody stId)) > $(← withNaryArg 3 (delabBody stId)))
1080+    | And _ _ =>
1081+      `($(← withNaryArg 0 (delabBody stId)) ∧ $(← withNaryArg 1 (delabBody stId)))
1082+    | Or _ _ =>
1083+      `($(← withNaryArg 0 (delabBody stId)) ∨ $(← withNaryArg 1 (delabBody stId)))
1084+    | Iff _ _ =>
1085+      `($(← withNaryArg 0 (delabBody stId)) ↔ $(← withNaryArg 1 (delabBody stId)))
1086+    | Not _ =>
1087+      `(¬ $(← withAppArg (delabBody stId)))
1088+    | _ =>
1089+      if e.isArrow then
1090+        `($(← withBindingDomain (delabBody stId)) →
1091+          $(← withBindingBody `h (delabBody stId)))
