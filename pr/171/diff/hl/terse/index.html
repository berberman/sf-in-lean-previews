53+-- Our goal in this chapter is to develop the tools to work
54+-- through some simple examples of *program verification* --
55+-- i.e., to use the precise definition of Imp to prove formally
56+-- that particular programs satisfy particular specifications

59+-- We'll develop a reasoning system called *Floyd-Hoare Logic*
60+-- -- often shortened to just *Hoare Logic* -- in which each of
61+-- the syntactic constructs of Imp is equipped with a generic
62+-- "proof rule" that can be used to reason compositionally

67+-- proof technique* for proving that programs are correct with
68+-- respect to such specifications -- where by "structured" we
69+-- mean that the structure of proofs directly mirrors the
70+-- structure of the programs that they are about.

446+-- Given two assertions `P` and `Q`, we say that `P` *implies*
447+-- `Q`, written `P ->> Q`, if, whenever `P` holds in some state
448+-- `st`, `Q` also holds.

472+-- ## Hoare Triples, Informally
473+
474+-- A *Hoare triple* is a claim about the state before and after
475+-- executing a command:
476+
477+--   {{P}} c {{Q}}
478+
479+-- This means:
480+
481+-- - If command `c` begins execution in a state satisfying
482+--   assertion `P`,
483+
484+-- - and if `c` eventually terminates in some final state,
485+
486+-- - then that final state will satisfy the assertion `Q`.
487+
488+-- Assertion `P` is called the *precondition* of the triple,
489+-- and `Q` is the *postcondition*.
490+
491+-- For example,
492+
493+-- - The Hoare triple
494+
495+--   {{X = 0}} X := X + 1 {{X = 1}}
496+
497+-- states that command `X := X + 1` will transform a state in
498+-- which `X = 0` to a state in which `X = 1`.
499+
500+-- - On the other hand,
501+
502+--   ∀ m, {{X = m}} X := X + 1 {{X = m + 1}}
503+
504+-- is a *proposition* stating that the Hoare triple
505+-- `{{X = m}} X :=
506+-- X + 1 {{X = m + 1}}` is valid for any choice
507+-- of `m`. Note that `m` in the two assertions is a reference
508+-- to the *Lean* variable `m`, which is bound outside the Hoare
509+-- triple.
510+

628+-- ## Hoare Triples, Formally
629+
630+-- We formalize valid Hoare triples in Lean as follows:
631+
632+def ValidHoareTriple
633+    (P : Assertion) (c : Com) (Q : Assertion) : Prop :=
634+  ∀ st st',
635+    (st =[ c ]=> st') →
636+    P st →
637+    Q st'
638+
639+-- Note to developers:
640+--     `NOTATION: SAZ 2024 One trickiness of these notations is that we
641+--     want the [com] and [assn] grammars to be "open", so that they can
642+--     include expressions parsed by the full [constr] grammar of Rocq.
643+--     However, then there is a conflict of precedence of the
644+--     "application" cases:
645+--
646+--     The example " {{ True }} X := 0 {{ False }} " does not parse as
647+--     intended because the [com] grammar includes the capability of
648+--     parsing the (ill-typed) term "0 {{ False }}".
649+--
650+--     This means that the "application" for [com] should disallow
651+--     arguments at the level at which the [assn] grammar is included
652+--     in constr.  The upshot is that this notation should be included
653+--     in the grammar at the *same* level as the assertion notation
654+--     [{{ P }}], which is 2.`

710+-- ## Proof Rules
711+
712+-- We want to be able to *prove* Hoare triples formally.
713+
714+-- Here's our plan:
715+
716+-- - introduce one "proof rule" for each Imp syntactic form
717+
718+-- - plus a couple of "structural rules" that help glue proofs
719+--   together
720+
721+-- - prove these rules correct in terms of the definition of
722+--   `ValidHoareTriple`
723+
724+-- - prove programs correct using these proof rules, without ever
725+--   unfolding the definition of `ValidHoareTriple`
726+
727+-- ### Skip
728+
729+-- Since `skip` doesn't change the state, it preserves any
730+-- assertion `P`:
731+
732+--   --------------------  (hoare_skip)
733+--   {{ P }} skip {{ P }}
734+
735+theorem hoare_skip (P : Assertion) :
736+    {{P}} skip; {{P}} := by
737+  intro st st' h hp
738+  inversion h
739+  assumption
740+
741+-- ### Sequencing
742+
743+-- If command `c1` takes any state where `P` holds to a state
744+-- where `Q` holds, and if `c2` takes any state where `Q` holds
745+-- to one where `R` holds, then doing `c1` followed by `c2`
746+-- will take any state where `P` holds to one where `R` holds:
747+
748+--    {{ P }} c1 {{ Q }}
749+--    {{ Q }} c2 {{ R }}
750+--   ----------------------  (hoare_seq)
751+--   {{ P }} c1;c2 {{ R }}
752+
753+theorem hoare_seq (P Q R : Assertion) (c1 c2 : Com)
754+    (h1 : {{Q}} ~c2 {{R}}) (h2 : {{P}} ~c1 {{Q}}) :
755+    {{P}} ~c1 ~c2 {{R}} := by
756+  intro st st' h12 pre
757+  inversion h12 with
758+  | seq st'' hc1 hc2 =>
759+    exact h1 _ _ hc2 (h2 _ _ hc1 pre)

761+-- ### Assignment
762+
763+-- How can we complete this triple?
764+
765+--   {{ ??? }}  X := Y  {{ X = 1 }}
766+
767+-- One natural possibility is:
768+
769+--   {{ Y = 1 }}  X := Y  {{ X = 1 }}
770+
771+-- The precondition is just the postcondition, but with `X`
772+-- replaced by `Y`.
773+
774+-- How about this one?
775+
776+--   {{ ??? }}  X := X + Y  {{ X = 1 }}
777+
778+-- Replace `X` with `X + Y`:
779+
780+--   {{ X + Y = 1 }}  X := X + Y  {{ X = 1 }}
781+
782+-- This works because "equals 1" holding of `X` is guaranteed
783+-- by the property "equals 1" holding of whatever is being
784+-- assigned to `X`.
785+
786+-- In general, the postcondition could be some arbitrary
787+-- assertion `Q`, and the right-hand side of the assignment
788+-- could be some arbitrary arithmetic expression `a`:
789+
790+--   {{ ??? }}  X := a  {{ Q }}
791+
792+-- The precondition would then be `Q`, but with any occurrences
793+-- of `X` in it replaced by `a`.
794+
795+-- Let's introduce a notation for this idea of replacing
796+-- occurrences: Define `Q \[X ↦ a`] to mean "`Q` where `a` is
797+-- substituted in place of `X`".
798+
799+-- This yields the Hoare logic rule for assignment:
800+
801+--   {{ Q [X ↦ a] }}  X := a  {{ Q }}
802+
803+-- One way of reading this rule is: If you want statement
804+-- `X := a` to terminate in a state that satisfies assertion
805+-- `Q`, then it suffices to start in a state that also
806+-- satisfies `Q`, except where `a` is substituted for every
807+-- occurrence of `X`.
808+
809+-- Here are some valid instances of the assignment rule:
810+
811+--   {{ (X ≤ 5) [X ↦ X + 1] }}         (that is, X + 1 ≤ 5)
812+--     X := X + 1
813+--   {{ X ≤ 5 }}
814+
815+--   {{ (X = 3) [X ↦ 3] }}              (that is, 3 = 3)
816+--     X := 3
817+--   {{ X = 3 }}
818+
819+--   {{ (0 ≤ X ∧ X ≤ 5) [X ↦ 3] }}.  (that is, 0 ≤ 3 ∧ 3 ≤ 5)
820+--     X := 3
821+--   {{ 0 ≤ X ∧ X ≤ 5 }}
822+
823+-- To formalize the rule, we must first formalize the idea of
824+-- "substituting an expression for an Imp variable in an
825+-- assertion", which we refer to as assertion substitution, or
826+-- `Assertion.sub`.
827+
828+-- Intuitively, given a proposition `P`, a variable `X`, and an
829+-- arithmetic expression `a`, we want to derive another
830+-- proposition `P'` that is just the same as `P` except that
831+-- `P'` should mention `a` wherever `P` mentions `X`.
832+
833+-- This operation is related to the idea of substituting Imp
834+-- expressions for Imp variables that we saw in *Equiv*
835+-- (`subst_aexp` and friends). The difference is that, here,
836+-- `P` is an arbitrary Lean assertion, so we can't directly
837+-- "edit" its text.
838+
839+-- However, we can achieve the same effect by evaluating `P` in
840+-- an updated state, defined as follows:
841+
842+def Assertion.sub (x : Ident) (a : Aexp) (P : Assertion) : Assertion :=
843+  fun (st : State) => P (x →ₜ a.eval st ; st)
844+
845+-- Note to developers (before next release):
846+--     This concrete syntax is hard to read in comments because
847+--     of all the square brackets. Something like
848+--     `P with X ↦ a` would be much better. I guess the same
849+--     will apply to the lambda-calculus chapters... BCP 25: I
850+--     still think this is a good idea, and I had a quick go at
851+--     implementing it, but did not succeed yet.
852+
853+/-- Assertion substitution: `P [X ↦ a]` -/
854+syntax:100 assn:100 " [" ident " ↦ " imp_aexp "]" : assn
855+
856+macro_rules
857+  | `({{ $P [$x ↦ $a] }}) => `(Assertion.sub $x (aexp { $a }) ({{ $P }}))
858+
859+-- This notation allows us to write this operation as:
860+
861+--   P [ X ↦ a ]
862+
863+-- That is, `P [X ↦ a]` stands for an assertion -- let's call
864+-- it `P'` -- that behaves just like `P` except that, wherever
865+-- `P` looks up the variable `X` in the current state, `P'`
866+-- instead uses the value of the expression `a`.
867+
868+-- We can demonstrate formally that we have captured intuitive
869+-- meaning of "assertion subsitution" by proving some example
870+-- logical equivalences:
871+
872+namespace ExampleAssertionSub
873+example :
874+    {{ (X ≤ 5) [X ↦ 3] }} <<->> {{ 3 ≤ 5 }} := by
875+  constructor <;> (unfold AssertImplies Assertion.sub; intro st h; exact h)
876+
877+example :
878+    {{ (X ≤ 5) [X ↦ X + 1] }} <<->> {{ (X + 1) ≤ 5 }} := by
879+  constructor <;> (unfold AssertImplies Assertion.sub; intro st h; exact h)
880+
881+end ExampleAssertionSub

1157+-- ### Consequence
1158+
1159+-- Sometimes the preconditions and postconditions we get from
1160+-- the Hoare rules won't quite be the ones we want in the
1161+-- particular situation at hand -- they may be logically
1162+-- equivalent but have a different syntactic form that fails to
1163+-- unify with the goal we are trying to prove, or they actually
1164+-- may be logically weaker (for preconditions) or stronger (for
1165+-- postconditions) than what we need.
1166+
1167+-- For instance,
1168+
1169+--   {{(X = 3) [X ↦ 3]}} X := 3 {{X = 3}},
1170+
1171+-- follows directly from the assignment rule, but
1172+
1173+--   {{True}} X := 3 {{X = 3}}
1174+
1175+-- does not. This triple is valid, but it is not an instance of
1176+-- `hoare_asgn` because `True` and `(X = 3) \[X ↦ 3`] are not
1177+-- syntactically equal assertions.
1178+
1179+-- However, they are logically *equivalent*, so if one triple
1180+-- is valid, then the other must certainly be as well. We can
1181+-- capture this observation with the following rule:
1182+
1183+--      {{P'}} c {{Q}}
1184+--        P <<->> P'
1185+--   ---------------------
1186+--      {{P}} c {{Q}}
1187+
1188+-- Taking this line of thought a bit further, we can see that
1189+-- strengthening the precondition or weakening the
1190+-- postcondition of a valid triple always produces another
1191+-- valid triple. This observation is captured by two *Rules of
1192+-- Consequence*.
1193+
1194+--          {{P'}} c {{Q}}
1195+--             P ->> P'
1196+--   -----------------------------   (hoare_consequence_pre)
1197+--          {{P}} c {{Q}}
1198+
1199+--          {{P}} c {{Q'}}
1200+--            Q' ->> Q
1201+--   -----------------------------    (hoare_consequence_post)
1202+--          {{P}} c {{Q}}
1203+
1204+-- Here are the formal versions:
1205+
1206+theorem hoare_consequence_pre (P P' Q : Assertion) (c : Com)
1207+    (hhoare : {{P'}} ~c {{Q}}) (himp : P ->> P') :
1208+    {{P}} ~c {{Q}} := by
1209+  unfold ValidHoareTriple
1210+  intro st st' heval hpre
1211+  apply hhoare st st'
1212+  · assumption
1213+  · apply himp
1214+    assumption
1215+
1216+theorem hoare_consequence_post (P Q Q' : Assertion) (c : Com)
1217+    (hhoare : {{P}} ~c {{Q'}}) (himp : Q' ->> Q) :
1218+    {{P}} ~c {{Q}} := by
1219+  unfold ValidHoareTriple
1220+  intro st st' heval hpre
1221+  apply himp
1222+  apply hhoare st st'
1223+  · assumption
1224+  · assumption
1225+
1226+-- For example, we can use the first consequence rule like
1227+-- this:
1228+
1229+--   {{ True }} ->>
1230+--   {{ (X = 1) [X ↦ 1] }}
1231+--     X := 1
1232+--   {{ X = 1 }}
1233+
1234+-- Or, formally...
1235+
1236+theorem hoare_asgn_example1 :
1237+    {{True}} X := 1; {{X = 1}} := by
1238+  sorry
1239+
1240+-- We can also use it to prove the example mentioned earlier.
1241+
1242+--   {{ X < 4 }} ->>
1243+--   {{ (X < 5)[X ↦ X + 1] }}
1244+--     X := X + 1
1245+--   {{ X < 5 }}
1246+
1247+-- Or, formally ...
1248+
1249+theorem assertion_sub_example2 :
1250+    {{X < 4}}
1251+      X := X + 1;
1252+    {{X < 5}} := by
1253+  sorry
1254+
1255+-- Finally, here is a combined rule of consequence that allows
1256+-- us to vary both the precondition and the postcondition.
1257+
1258+--          {{P'}} c {{Q'}}
1259+--             P ->> P'
1260+--             Q' ->> Q
1261+--   -----------------------------   (hoare_consequence)
1262+--          {{P}} c {{Q}}
1263+
1264+theorem hoare_consequence (P P' Q Q' : Assertion) (c : Com)
1265+    (htriple : {{P'}} ~c {{Q'}}) (hpre : P ->> P') (hpost : Q' ->> Q) :
1266+    {{P}} ~c {{Q}} := by
1267+  apply hoare_consequence_pre (P' := P')
1268+  · apply hoare_consequence_post (Q' := Q')
1269+    · assumption
1270+    · assumption
1271+  · assumption

1452+-- ### Conditionals
1453+
1454+-- What sort of rule do we want for reasoning about conditional
1455+-- commands?
1456+
1457+-- Certainly, if the same assertion `Q` holds after executing
1458+-- either of the branches, then it holds after the whole
1459+-- conditional. So we might be tempted to write:
1460+
1461+--           {{P}} c1 {{Q}}
1462+--           {{P}} c2 {{Q}}
1463+--   ---------------------------------
1464+--   {{P}} if b then c1 else c2 {{Q}}
1465+
1466+-- However, this is rather weak. For example, using this rule,
1467+-- we cannot show
1468+
1469+--   {{ True }}
1470+--     if X = 0
1471+--       then Y := 2
1472+--       else Y := X + 1
1473+--     end
1474+--   {{ X ≤ Y }}
1475+
1476+-- since the rule doesn't tell us enough about the state in
1477+-- which the assignments take place in the "then" and "else"
1478+-- branches.
1479+
1480+-- Better:
1481+
1482+--   {{P ∧   b}} c1 {{Q}}
1483+--   {{P ∧ ¬ b}} c2 {{Q}}
1484+--   ------------------------------------  (hoare_if)
1485+--   {{P}} if b then c1 else c2 end {{Q}}
1486+
1487+-- To make this formal, we need a way of formally "lifting" any
1488+-- bexp `b` to an assertion.
1489+
1490+-- We'll write `bassertion b` for the assertion "the boolean
1491+-- expression `b` evaluates to `true`."
1492+
1493+def bassertion (b : Bexp) : Assertion :=
1494+  fun st => (b.eval st = true)
1495+
1496+@[simp] theorem bassertion_apply (b : Bexp) (st : State) :
1497+    bassertion b st = (b.eval st = true) := rfl

1642+-- ### While Loops
1643+
1644+-- The Hoare rule for `while` loops is based on the idea of a
1645+-- *command invariant* (or just *invariant*): an assertion
1646+-- whose truth is guaranteed after executing a command,
1647+-- assuming it is true before.
1648+
1649+-- That is, an assertion `P` is a command invariant of `c` if
1650+
1651+--   {{P}} c {{P}}
1652+
1653+-- holds. Note that the command invariant might temporarily
1654+-- become false in the middle of executing `c`, but by the end
1655+-- of `c` it must be restored.
1656+
1657+-- The Hoare while rule combines the idea of a command
1658+-- invariant with information about when guard `b` does or does
1659+-- not hold.
1660+
1661+--         {{P ∧ b}} c {{P}}
1662+--   --------------------------------- (hoare_while)
1663+--   {{P}} while b do c end {{P ∧ ¬b}}
1664+
1665+-- Note to developers:
1666+--     HIDE: The big comment will not display nicely. But I
1667+--     guess it's folded...
1668+
1669+theorem hoare_while (P : Assertion) (b : Bexp) (c : Com)
1670+    (hhoare : {{P ∧ b}} ~c {{P}}) :
1671+    {{P}} while (~b) { ~c } {{P ∧ ¬b}} := by
1672+  intro st st' heval hP
1673+  /- We proceed by induction on `heval`, because, in the "keep
1674+  looping" case, its hypotheses talk about the whole loop instead
1675+  of just `c`. The auxiliary statement `key` generalizes over an
1676+  arbitrary command, together with an equation remembering that the

48+  This function is similar to `forallMetaTelescopeReducing`: Given `e` of the
49+  form `forall ..xs, A`, this combinator will create a new metavariable for
50+  each `x` in `xs` until it reaches an `x` whose type is defeq to `t`,
51+  and instantiate `A` with these, while also reducing `A` if needed.
52+  It uses `forallMetaTelescopeReducing`.
53+
54+  This function returns a triple `(mvs, bis, out)` where
55+  - `mvs` is an array containing the new metavariables.
56+  - `bis` is an array containing the binder infos for the `mvs`.
57+  - `out` is `e` but instantiated with the `mvs`.

59+  Lifted from [https://github.com/leanprover-community/mathlib4/blob/master/Mathlib/Lean/Meta/Basic.lean#L41].

61+def forallMetaTelescopeReducingUntilDefEq
62+    (e t : Expr) (kind : MetavarKind := MetavarKind.natural) :
63+    MetaM (Array Expr × Array BinderInfo × Expr) := do
64+  let (ms, bs, tp) ← forallMetaTelescopeReducing e (some 1) kind
65+  unless ms.size == 1 do
66+    if ms.size == 0 then throwError m!"Failed: {← ppExpr e} is not the type of a function."
67+    else throwError m!"Failed"
68+  let mut mvs := ms
69+  let mut bis := bs
70+  let mut out : Expr := tp
71+  while !(← isDefEq (← inferType mvs.toList.getLast!) t) do
72+    let (ms, bs, tp) ← forallMetaTelescopeReducing out (some 1) kind
73+    unless ms.size == 1 do
74+      throwError m!"Failed to find {← ppExpr t} as the type of a parameter of {← ppExpr e}."
75+    mvs := mvs ++ ms
76+    bis := bis ++ bs
77+    out := tp
78+  return (mvs, bis, out)

190+/-- Given an inversion alternative and a list of goals,
191+  solve the tagged goal with the provided tactics,
192+  throwing an error if the goal cannot be found or solved. -/
193+def evalInvAlt (goals : List CasesSubgoal) (alt : TSyntax `invAlt) : TacticM (List MVarId) :=
194+  match alt with
195+  | `(invAlt| | $tag:ident $vars:binderIdent* => $tactics:tacticSeq) => do
196+    if let some goal := goals.find? (matchTag tag.getId $ ·.ctorName.getD default) then
197+      return [← trySolveGoal vars tactics goal.mvarId]
198+    else throwError m!"Invalid alternative name `{tag.getId}`: {← errorMsg}"
199+  | `(invAlt| | _ $vars:binderIdent* => $tactics:tacticSeq) => do
200+    if !goals.isEmpty then
201+      goals.mapM (trySolveGoal vars tactics ·.mvarId)
202+    else throwErrorAt alt m!"Invalid wildcard alternative: {← errorMsg}"
203+  | _ => throwErrorAt alt "Could not parse inversion alternative"

205+  matchTag tag name : Bool :=
206+    tag == name || tag.isPrefixOf name || tag.isSuffixOf name
207+  trySolveGoal vars tactics goal : TacticM MVarId := do
208+    let goals ← renameInaccessibles goal vars >>= evalTacticAt tactics
209+    unless goals.isEmpty do
210+      reportUnsolvedGoals goals
211+    return goal
212+  errorMsg : TacticM MessageData := do
213+    let goalMsgs := goals.filterMap (.ofName <$> ·.ctorName)
214+    return if goals.isEmpty
215+      then m!"There are no unhandled alternatives"
216+      else m!"Expected {.orList goalMsgs}"
