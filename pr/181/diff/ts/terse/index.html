30+-- THE SIMPLY TYPED LAMBDA CALCULUS
31+
32+-- Syntax:
33+
34+-- t ::= x                     (variable)
35+--     | λ x : T . t           (abstraction)
36+--     | t t                   (application)
37+--     | true                  (constant true)
38+--     | false                 (constant false)
39+--     | if t then t else t    (conditional)
40+
41+-- Values:
42+
43+-- v ::= λ x : T . t
44+--     | true
45+--     | false
46+
47+-- Substitution:
48+
49+--   [x:=s]x               = s
50+--   [x:=s]y               = y                     if x ≠ y
51+--   [x:=s](λx:T. t)       = λx:T. t
52+--   [x:=s](λy:T. t)       = λy:T. [x:=s]t         if x ≠ y
53+--   [x:=s](t₁ t₂)         = ([x:=s]t₁) ([x:=s]t₂)
54+--   [x:=s]true            = true
55+--   [x:=s]false           = false
56+--   [x:=s](if t₁ then t₂ else t₃) =
57+--                   if [x:=s]t₁ then [x:=s]t₂ else [x:=s]t₃
58+
59+-- Small-step operational semantics:
60+
61+-- v.IsValue
62+--                        -----------------------                    (appAbs)
63+--                         (λx:T. t) v ⟶ [x:=v]t
64+
65+--                               t₁ ⟶ t₁'
66+--                           ----------------                        (app1)
67+--                            t₁ t₂ ⟶ t₁' t₂
68+
69+--                              v₁.IsValue
70+--                               t₂ ⟶ t₂'
71+--                           ----------------                        (app2)
72+--                            v₁ t₂ ⟶ v₁ t₂'
73+
74+--                   --------------------------------                (ifTrue)
75+--                    (if true then t₁ else t₂) ⟶ t₁
76+
77+--                   ---------------------------------               (ifFalse)
78+--                    (if false then t₁ else t₂) ⟶ t₂
79+
80+--                               t₁ ⟶ t₁'
81+--         ----------------------------------------------------      (ifStep)
82+--          (if t₁ then t₂ else t₃) ⟶ (if t₁' then t₂ else t₃)
83+
84+-- Typing:
85+
86+-- Γ x = T₁
87+--                             ------------                       (var)
88+--                              Γ ⊢ x ⦂ T₁
89+
90+--                         x ↦ T₂ ; Γ ⊢ t₁ ⦂ T₁
91+--                       -------------------------                (abs)
92+--                        Γ ⊢ λx:T₂. t₁ ⦂ T₂ → T₁
93+
94+--                           Γ ⊢ t₁ ⦂ T₂ → T₁
95+--                             Γ ⊢ t₂ ⦂ T₂
96+--                          ------------------                    (app)
97+--                            Γ ⊢ t₁ t₂ ⦂ T₁
98+
99+--                           -----------------                    (tru)
100+--                            Γ ⊢ true ⦂ Bool
101+
102+--                          ------------------                    (fls)
103+--                           Γ ⊢ false ⦂ Bool
104+
105+--              Γ ⊢ t₁ ⦂ Bool    Γ ⊢ t₂ ⦂ T₁    Γ ⊢ t₃ ⦂ T₁
106+--             ---------------------------------------------      (ite)
107+--                    Γ ⊢ if t₁ then t₂ else t₃ ⦂ T₁
108+

125+-- ## Canonical Forms
126+
127+-- Formally, we will need these lemmas only for terms that are
128+-- not only well typed but *closed* — i.e., well typed in the
129+-- empty context.
130+
131+theorem canonical_forms_bool (t : Tm) (hT : <{ ∅ ⊢ ~t ⦂ Bool }>) (hv : t.IsValue) :
132+    t = <{ true }> ∨ t = <{ false }> := by
133+  cases hv with
134+  | abs x T t₁ => cases hT
135+  | tru => exact .inl rfl
136+  | fls => exact .inr rfl
137+
138+theorem canonical_forms_fun (t : Tm) (T₁ T₂ : Ty)
139+    (hT : <{ ∅ ⊢ ~t ⦂ ~T₁ → ~T₂ }>) (hv : t.IsValue) :
140+    ∃ x u, t = <{ λ ~x : ~T₁ . ~u }> := by
141+  cases hv with
142+  | abs x T t₁ => cases hT with | abs _ _ _ _ _ _ => exact ⟨x, t₁, rfl⟩
143+  | tru => cases hT
144+  | fls => cases hT

146+-- ## Progress
147+
148+-- The *progress* theorem tells us that closed, well-typed
149+-- terms are not stuck.
150+
151+theorem progress (t : Tm) (T : Ty) (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) :
152+    t.IsValue ∨ ∃ t', t ⟶ t' := by
153+  generalize hΓ : (∅ : Context) = Γ at hT
154+  induction hT with
155+  | var Γ x T₁ h =>
156+    subst hΓ
157+    -- Contradictory: variables cannot be typed in an empty context.
158+    rw [PartialMap.apply_empty] at h
159+    cases h
160+  | abs => exact .inl (.abs ..)
161+  | tru => exact .inl .tru
162+  | fls => exact .inl .fls
163+  | app Γ T₁ T₂ t₁ t₂ h₁ h₂ ih₁ ih₂ =>
164+    -- `t = t₁ t₂`.  Proceed by cases on whether `t₁` is a value or steps.
165+    right
166+    cases ih₁ hΓ with
167+    | inl hv₁ =>
168+      cases ih₂ hΓ with
169+      | inl hv₂ =>
170+        obtain ⟨x, u, rfl⟩ := canonical_forms_fun t₁ _ _ (hΓ ▸ h₁) hv₁
171+        exact ⟨<{ [~x := ~t₂] ~u }>, .appAbs x T₂ u t₂ hv₂⟩
172+      | inr hs₂ => obtain ⟨t₂', h⟩ := hs₂; exact ⟨<{ ~t₁ ~t₂' }>, .app2 t₁ t₂ t₂' hv₁ h⟩
173+    | inr hs₁ => obtain ⟨t₁', h⟩ := hs₁; exact ⟨<{ ~t₁' ~t₂ }>, .app1 t₁ t₁' t₂ h⟩
174+  | ite Γ t₁ t₂ t₃ T₁ h₁ h₂ h₃ ih₁ ih₂ ih₃ =>
175+    right
176+    cases ih₁ hΓ with
177+    | inl hv₁ =>
178+      cases canonical_forms_bool t₁ (hΓ ▸ h₁) hv₁ with
179+      | inl he => subst he; exact ⟨t₂, .ifTrue t₂ t₃⟩
180+      | inr he => subst he; exact ⟨t₃, .ifFalse t₂ t₃⟩
181+    | inr hs₁ =>
182+      obtain ⟨t₁', h⟩ := hs₁
183+      exact ⟨<{ if ~t₁' then ~t₂ else ~t₃ }>, .ifStep t₁ t₁' t₂ t₃ h⟩

185+-- ## Preservation
186+
187+-- For preservation, we need some technical machinery for
188+-- reasoning about variables and substitution.
189+
190+-- - The *preservation theorem* is proved by induction on a
191+--   typing derivation and case analysis on the step relation,
192+--   pretty much as we did in the Types chapter.
193+
194+--   Main novelty: `Step.appAbs` uses the substitution operation.
195+
196+--   To see that this step preserves typing, we need to know that
197+--   the substitution itself does. So we prove a...
198+
199+-- - *substitution lemma*, stating that substituting a (closed,
200+--   well-typed) term `s` for a variable `x` in a term `t`
201+--   preserves the type of `t`.
202+
203+-- The proof goes by induction on the form of `t` and requires
204+-- looking at all the different cases in the definition of
205+-- substitition.
206+
207+-- Tricky case: variables.
208+
209+-- In this case, we need to deduce from the fact that a term
210+-- `s` has type S in the empty context the fact that `s` has
211+-- type S in every context.
212+
213+-- For this we prove a...
214+
215+-- - *weakening* lemma, showing that typing is preserved under
216+--   "extensions" to the context `Γ`.
217+
218+-- To make Lean happy, we need to formalize all this in the
219+-- opposite order...
220+
221+-- ### The Weakening Lemma
222+
223+-- First, we show that typing is preserved under "extensions"
224+-- to the context `Γ`. (Recall map inclusion, `Γ ⊆ Γ'`, from
225+-- the `Typeclasses` chapter.)
226+
227+theorem weakening (Γ Γ' : Context) (t : Tm) (T : Ty)
228+    (hi : Γ ⊆ Γ') (hT : <{ ~Γ ⊢ ~t ⦂ ~T }>) : <{ ~Γ' ⊢ ~t ⦂ ~T }> := by
229+  induction hT generalizing Γ' with
230+  | var _ x _ h => exact .var _ x _ (hi x _ h)
231+  | abs _ x _ _ _ _ ih => exact .abs _ x _ _ _ (ih _ (PartialMap.update_subset _ _ _ _ hi))
232+  | app _ _ _ _ _ _ _ ih₁ ih₂ => exact .app _ _ _ _ _ (ih₁ _ hi) (ih₂ _ hi)
233+  | tru => exact .tru _
234+  | fls => exact .fls _
235+  | ite _ _ _ _ _ _ _ _ ih₁ ih₂ ih₃ => exact .ite _ _ _ _ _ (ih₁ _ hi) (ih₂ _ hi) (ih₃ _ hi)
236+
237+-- The following simple corollary is what we actually need
238+-- below.
239+
240+theorem weakening_empty (Γ : Context) (t : Tm) (T : Ty) (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) :
241+    <{ ~Γ ⊢ ~t ⦂ ~T }> :=
242+  weakening _ _ _ _
243+    (fun x T h => by rw [PartialMap.apply_empty] at h; cases h) hT
244+
245+-- ### The Substitution Lemma
246+
247+-- Now we come to the conceptual heart of the proof that
248+-- reduction preserves types — namely, the observation that
249+-- *substitution* preserves types.
250+
251+-- The *substitution lemma* says:
252+
253+-- - Suppose we have a term `t` with a free variable `x`, and
254+--   suppose we've been able to assign a type `T` to `t` under
255+--   the assumption that `x` has some type `U`.
256+
257+-- - Also, suppose that we have some other term `v` and that
258+--   we've shown that `v` has type `U`.
259+
260+-- - Then we can substitute `v` for each of the occurrences of
261+--   `x` in `t` and obtain a new term that still has type `T`.
262+
263+theorem substitution_preserves_typing (Γ : Context) (x : String) (U : Ty)
264+    (t v : Tm) (T : Ty)
265+    (hT : <{ ~x ↦ ~U ; ~Γ ⊢ ~t ⦂ ~T }>) (hv : <{ ∅ ⊢ ~v ⦂ ~U }>) :
266+    <{ ~Γ ⊢ [~x := ~v] ~t ⦂ ~T }> := by
267+  -- By induction on `t`; in each case we get at the derivation of `hT`.
268+  induction t generalizing Γ T with
269+  | var y =>
270+    cases hT with
271+    | var _ _ _ h =>
272+      by_cases hxy : x = y
273+      · subst hxy
274+        rw [PartialMap.update_eq] at h
275+        rw [subst_var_eq]
276+        have hUT : U = T := Option.some.inj h
277+        subst hUT
278+        exact weakening_empty _ _ _ hv
279+      · rw [PartialMap.update_neq _ _ _ hxy] at h
280+        rw [subst_var_ne _ _ _ hxy]
281+        exact .var _ y _ h
282+  | app t₁ t₂ ih₁ ih₂ =>
283+    cases hT with
284+    | app _ _ _ _ _ h₁ h₂ => rw [subst_app]; exact .app _ _ _ _ _ (ih₁ _ _ h₁) (ih₂ _ _ h₂)
285+  | abs y S t₁ ih =>
286+    cases hT with
287+    | abs _ _ _ _ _ h =>
288+      by_cases hxy : x = y
289+      · subst hxy
290+        rw [subst_abs_eq]
291+        rw [PartialMap.update_shadow] at h
292+        exact .abs _ _ _ _ _ h
293+      · rw [subst_abs_ne _ _ _ _ _ hxy]
294+        rw [PartialMap.update_permute _ _ _ _ _ (Ne.symm hxy)] at h
295+        exact .abs _ _ _ _ _ (ih _ _ h)
296+  | tru => cases hT with | tru => rw [subst_tru]; exact .tru _
297+  | fls => cases hT with | fls => rw [subst_fls]; exact .fls _
298+  | ite c t e ihc iht ihe =>
299+    cases hT with
300+    | ite _ _ _ _ _ h₁ h₂ h₃ =>
301+      rw [subst_ite]
302+      exact .ite _ _ _ _ _ (ihc _ _ h₁) (iht _ _ h₂) (ihe _ _ h₃)
303+
304+-- ### Main Theorem
305+
306+-- We now have the ingredients we need to prove preservation:
307+-- if a closed, well-typed term `t` has type `T` and takes a
308+-- step to `t'`, then `t'` is also a closed term with type `T`.
309+-- In other words, the small-step reduction relation preserves
310+-- types.
311+
312+theorem preservation (t t' : Tm) (T : Ty)
313+    (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) (hs : t ⟶ t') : <{ ∅ ⊢ ~t' ⦂ ~T }> := by
314+  generalize hΓ : (∅ : Context) = Γ at hT
315+  induction hT generalizing t' with
316+  | var => cases hs
317+  | abs => cases hs
318+  | tru => cases hs
319+  | fls => cases hs
320+  | app Γ T₁ T₂ t₁ t₂ h₁ h₂ ih₁ ih₂ =>
321+    subst hΓ
322+    cases hs with
323+    | appAbs _ _ _ _ _ =>
324+      -- The one interesting case: the desired result is the substitution lemma.
325+      cases h₁ with
326+      | abs _ _ _ _ _ hb => exact substitution_preserves_typing _ _ _ _ _ _ hb h₂
327+    | app1 _ t₁' _ h => exact .app _ _ _ _ _ (ih₁ t₁' h rfl) h₂
328+    | app2 _ _ t₂' _ h => exact .app _ _ _ _ _ h₁ (ih₂ t₂' h rfl)
329+  | ite Γ t₁ t₂ t₃ T₁ h₁ h₂ h₃ ih₁ ih₂ ih₃ =>
330+    subst hΓ
331+    cases hs with
332+    | ifTrue => exact h₂
333+    | ifFalse => exact h₃
334+    | ifStep _ t₁' _ _ h => exact .ite _ _ _ _ _ (ih₁ t₁' h rfl) h₂ h₃
335+
336+end Stlc

338+-- Let's extend the STLC with a base type of numbers, some
339+-- constants, and some primitive operators.
340+
341+namespace StlcArith
342+
343+-- To types, we add a base type of natural numbers (and remove
344+-- booleans, for brevity).
345+
346+inductive Ty where
347+  | arrow (T₁ T₂ : Ty)
348+  | nat
349+
350+-- To terms, we add natural number constants, along with
351+-- successor, predecessor, multiplication, and zero-testing.
352+
353+inductive Tm where
354+  | var (x : String)
355+  | app (t₁ t₂ : Tm)
356+  | abs (x : String) (T : Ty) (t : Tm)
357+  | const (n : Nat)
358+  | succ (t : Tm)
359+  | pred (t : Tm)
360+  | mult (t₁ t₂ : Tm)
361+  | ite0 (c t e : Tm)
362+
363+-- _Details:_ Notation encoding: types
364+
365+-- The type grammar needs no new productions: `Nat` is a bare
366+-- identifier, which the template already accepts, and arrows
367+-- and parentheses are unchanged. Only the `macro_rules` are
368+-- new, and they differ from the STLC's in just two places —
369+-- the identifier `Nat` names this language's base type, and
370+-- the arrow builds this language's `StlcArith.Ty.arrow`.
371+
372+scoped macro_rules (kind := Stlc.tyBracket)
373+  | `(<{ ~$T:term }>)    => pure T
374+  | `(<{ ($T:stlcTy) }>) => `(<{ $T:stlcTy }>)
375+  | `(<{ $x:ident }>) =>
376+      match x.getId.toString with
377+      | "Nat" => `(Ty.nat)
378+      | _ => `(($x : Ty))
379+  | `(<{ $T₁:stlcTy → $T₂:stlcTy }>)  => `(Ty.arrow <{ $T₁:stlcTy }> <{ $T₂:stlcTy }>)
380+  | `(<{ $T₁:stlcTy -> $T₂:stlcTy }>) => `(Ty.arrow <{ $T₁:stlcTy }> <{ $T₂:stlcTy }>)
381+
382+-- _Details:_ Notation encoding: terms
383+
384+-- Terms do need new productions: a numeral, an infix `*`, and
385+-- the zero test. Multiplication binds looser than application
386+-- and tighter than `λ`, so `x * y z` multiplies `x` by the
387+-- application `y z`; it associates to the right, so
388+-- `x * y * z` is `x * (y * z)`.
389+
390+-- `succ` and `pred` get no production of their own. Making
391+-- them keywords would reserve those words globally — and we
392+-- would then be unable to write `succ` as a case name in a
393+-- proof, including for Lean's own `Nat`. Instead they are
394+-- written as though they were functions applied to an
395+-- argument, `succ t`, and the application rule below
396+-- recognizes them. `if0` **is** a keyword, since `then` and
397+-- `else` leave no other option; that is why the constructor
398+-- above is called `StlcArith.Tm.ite0` rather than `if0`, just
399+-- as the STLC's conditional is `Stlc.Tm.ite`.
400+
401+scoped syntax:max num : stlcTm
402+scoped syntax:60 stlcTm:61 " * " stlcTm:60 : stlcTm
403+scoped syntax:50 "if0 " stlcTm:51 " then " stlcTm:50 " else " stlcTm:50 : stlcTm
404+
405+open Lean in
406+scoped macro_rules (kind := Stlc.tmBracket)
407+  | `(<{ ~$e:term }>)    => pure e
408+  | `(<{ ($t:stlcTm) }>) => `(<{ $t:stlcTm }>)
409+  | `(<{ $n:num }>)      => `(Tm.const $n)
410+  | `(<{ $x:ident }>) =>
411+      match x.getId.toString with
412+      | "Nat"  => Macro.throwErrorAt x "`Nat` is a type, not a term"
413+      | "succ" => Macro.throwErrorAt x "`succ` must be applied to an argument"
414+      | "pred" => Macro.throwErrorAt x "`pred` must be applied to an argument"
415+      | _      => `(Tm.var $(quote x.getId.toString))
416+  | `(<{ $t₁:stlcTm $t₂:stlcTm }>) =>
417+      match t₁ with
418+      | `(stlcTm| $f:ident) =>
419+          match f.getId.toString with
420+          | "succ" => `(Tm.succ <{ $t₂:stlcTm }>)
421+          | "pred" => `(Tm.pred <{ $t₂:stlcTm }>)
422+          | _      => `(Tm.app <{ $t₁:stlcTm }> <{ $t₂:stlcTm }>)
423+      | _ => `(Tm.app <{ $t₁:stlcTm }> <{ $t₂:stlcTm }>)
424+  | `(<{ λ $x : $T . $t }>) => do
425+      `(Tm.abs $(← Stlc.varStr x) <{ $T:stlcTy }> <{ $t:stlcTm }>)
426+  | `(<{ $t₁:stlcTm * $t₂:stlcTm }>) => `(Tm.mult <{ $t₁:stlcTm }> <{ $t₂:stlcTm }>)
427+  | `(<{ if0 $c then $t else $e }>) =>
428+      `(Tm.ite0 <{ $c:stlcTm }> <{ $t:stlcTm }> <{ $e:stlcTm }>)
429+
430+-- _Details:_ Notation encoding: printing it back
431+
432+-- As in the Stlc chapter, a delaborator runs the grammar
433+-- backwards, so that goals mentioning these terms and types
434+-- read in the concrete syntax. The parenthesizers registered
435+-- there are for the whole syntax category, so they serve this
436+-- language too and are not repeated.
437+
438+open Lean PrettyPrinter Delaborator SubExpr in
439+/-- Rebuild `stlcTy` concrete syntax from a `Ty` value. -/
440+partial def delabTyInner : DelabM (TSyntax `stlcTy) := do
441+  let stx ←
442+    match_expr ← getExpr with
443+    | Ty.nat => `(stlcTy| $(mkIdent `Nat):ident)
444+    | Ty.arrow _ _ => do
445+        let a ← withAppFn <| withAppArg delabTyInner
446+        let b ← withAppArg delabTyInner
447+        `(stlcTy| $a → $b)
448+    | _ => do
449+        match ← delab with
450+        | `($i:ident) => `(stlcTy| $i:ident)
451+        | e => `(stlcTy| ~$e)
452+  (⟨·⟩) <$> annotateTermInfo ⟨stx.raw⟩
453+
454+open Lean PrettyPrinter Delaborator SubExpr in
455+/-- Rebuild `stlcTm` concrete syntax from a `Tm` value. -/
456+partial def delabTmInner : DelabM (TSyntax `stlcTm) := do
457+  let stx ←
458+    match_expr ← getExpr with
459+    | Tm.var _ => do
460+        match ← withAppArg delab with
461+        | `($s:str) =>
462+            if Stlc.isPlainName s.getString then
463+              `(stlcTm| $(mkIdent (Name.mkSimple s.getString)):ident)
464+            else `(stlcTm| ~($(⟨← delab⟩)))
465+        | _ => `(stlcTm| ~($(⟨← delab⟩)))
466+    | Tm.const _ => do
467+        match ← withAppArg delab with
468+        | `($n:num) => `(stlcTm| $n:num)
469+        | _ => `(stlcTm| ~($(⟨← delab⟩)))
470+    | Tm.app _ _ => do
471+        let f ← withAppFn <| withAppArg delabTmInner
472+        let a ← withAppArg delabTmInner
473+        `(stlcTm| $f $a)
474+    | Tm.abs _ _ _ => do
475+        let x ← withAppFn <| withAppFn <| withAppArg Stlc.delabVarInner
476+        let T ← withAppFn <| withAppArg delabTyInner
477+        let t ← withAppArg delabTmInner
