30added-- THE SIMPLY TYPED LAMBDA CALCULUS
31added
32added-- Syntax:
33added
34added-- t ::= x (variable)
35added-- | λ x : T . t (abstraction)
36added-- | t t (application)
37added-- | true (constant true)
38added-- | false (constant false)
39added-- | if t then t else t (conditional)
40added
41added-- Values:
42added
43added-- v ::= λ x : T . t
44added-- | true
45added-- | false
46added
47added-- Substitution:
48added
49added-- [x:=s]x = s
50added-- [x:=s]y = y if x ≠ y
51added-- [x:=s](λx:T. t) = λx:T. t
52added-- [x:=s](λy:T. t) = λy:T. [x:=s]t if x ≠ y
53added-- [x:=s](t₁ t₂) = ([x:=s]t₁) ([x:=s]t₂)
54added-- [x:=s]true = true
55added-- [x:=s]false = false
56added-- [x:=s](if t₁ then t₂ else t₃) =
57added-- if [x:=s]t₁ then [x:=s]t₂ else [x:=s]t₃
58added
59added-- Small-step operational semantics:
60added
61added-- v.IsValue
62added-- ----------------------- (appAbs)
63added-- (λx:T. t) v ⟶ [x:=v]t
64added
65added-- t₁ ⟶ t₁'
66added-- ---------------- (app1)
67added-- t₁ t₂ ⟶ t₁' t₂
68added
69added-- v₁.IsValue
70added-- t₂ ⟶ t₂'
71added-- ---------------- (app2)
72added-- v₁ t₂ ⟶ v₁ t₂'
73added
74added-- -------------------------------- (ifTrue)
75added-- (if true then t₁ else t₂) ⟶ t₁
76added
77added-- --------------------------------- (ifFalse)
78added-- (if false then t₁ else t₂) ⟶ t₂
79added
80added-- t₁ ⟶ t₁'
81added-- ---------------------------------------------------- (ifStep)
82added-- (if t₁ then t₂ else t₃) ⟶ (if t₁' then t₂ else t₃)
83added
84added-- Typing:
85added
86added-- Γ x = T₁
87added-- ------------ (var)
88added-- Γ ⊢ x ⦂ T₁
89added
90added-- x ↦ T₂ ; Γ ⊢ t₁ ⦂ T₁
91added-- ------------------------- (abs)
92added-- Γ ⊢ λx:T₂. t₁ ⦂ T₂ → T₁
93added
94added-- Γ ⊢ t₁ ⦂ T₂ → T₁
95added-- Γ ⊢ t₂ ⦂ T₂
96added-- ------------------ (app)
97added-- Γ ⊢ t₁ t₂ ⦂ T₁
98added
99added-- ----------------- (tru)
100added-- Γ ⊢ true ⦂ Bool
101added
102added-- ------------------ (fls)
103added-- Γ ⊢ false ⦂ Bool
104added
105added-- Γ ⊢ t₁ ⦂ Bool Γ ⊢ t₂ ⦂ T₁ Γ ⊢ t₃ ⦂ T₁
106added-- --------------------------------------------- (ite)
107added-- Γ ⊢ if t₁ then t₂ else t₃ ⦂ T₁
108added

146added-- ## Progress
147added
148added-- The *progress* theorem tells us that closed, well-typed
149added-- terms are not stuck.
150added
151addedtheorem progress (t : Tm) (T : Ty) (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) :
152added t.IsValue ∨ ∃ t', t ⟶ t' := by
153added generalize hΓ : (∅ : Context) = Γ at hT
154added induction hT with
155added | var Γ x T₁ h =>
156added subst hΓ
157added -- Contradictory: variables cannot be typed in an empty context.
158added rw [PartialMap.apply_empty] at h
159added cases h
160added | abs => exact .inl (.abs ..)
161added | tru => exact .inl .tru
162added | fls => exact .inl .fls
163added | app Γ T₁ T₂ t₁ t₂ h₁ h₂ ih₁ ih₂ =>
164added -- `t = t₁ t₂`. Proceed by cases on whether `t₁` is a value or steps.
165added right
166added cases ih₁ hΓ with
167added | inl hv₁ =>
168added cases ih₂ hΓ with
169added | inl hv₂ =>
170added obtain ⟨x, u, rfl⟩ := canonical_forms_fun t₁ _ _ (hΓ ▸ h₁) hv₁
171added exact ⟨<{ [~x := ~t₂] ~u }>, .appAbs x T₂ u t₂ hv₂⟩
172added | inr hs₂ => obtain ⟨t₂', h⟩ := hs₂; exact ⟨<{ ~t₁ ~t₂' }>, .app2 t₁ t₂ t₂' hv₁ h⟩
173added | inr hs₁ => obtain ⟨t₁', h⟩ := hs₁; exact ⟨<{ ~t₁' ~t₂ }>, .app1 t₁ t₁' t₂ h⟩
174added | ite Γ t₁ t₂ t₃ T₁ h₁ h₂ h₃ ih₁ ih₂ ih₃ =>
175added right
176added cases ih₁ hΓ with
177added | inl hv₁ =>
178added cases canonical_forms_bool t₁ (hΓ ▸ h₁) hv₁ with
179added | inl he => subst he; exact ⟨t₂, .ifTrue t₂ t₃⟩
180added | inr he => subst he; exact ⟨t₃, .ifFalse t₂ t₃⟩
181added | inr hs₁ =>
182added obtain ⟨t₁', h⟩ := hs₁
183added exact ⟨<{ if ~t₁' then ~t₂ else ~t₃ }>, .ifStep t₁ t₁' t₂ t₃ h⟩

185added-- ## Preservation
186added
187added-- For preservation, we need some technical machinery for
188added-- reasoning about variables and substitution.
189added
190added-- - The *preservation theorem* is proved by induction on a
191added-- typing derivation and case analysis on the step relation,
192added-- pretty much as we did in the Types chapter.
193added
194added-- Main novelty: `Step.appAbs` uses the substitution operation.
195added
196added-- To see that this step preserves typing, we need to know that
197added-- the substitution itself does. So we prove a...
198added
199added-- - *substitution lemma*, stating that substituting a (closed,
200added-- well-typed) term `s` for a variable `x` in a term `t`
201added-- preserves the type of `t`.
202added
203added-- The proof goes by induction on the form of `t` and requires
204added-- looking at all the different cases in the definition of
205added-- substitition.
206added
207added-- Tricky case: variables.
208added
209added-- In this case, we need to deduce from the fact that a term
210added-- `s` has type S in the empty context the fact that `s` has
211added-- type S in every context.
212added
213added-- For this we prove a...
214added
215added-- - *weakening* lemma, showing that typing is preserved under
216added-- "extensions" to the context `Γ`.
217added
218added-- To make Lean happy, we need to formalize all this in the
219added-- opposite order...
220added
221added-- ### The Weakening Lemma
222added
223added-- First, we show that typing is preserved under "extensions"
224added-- to the context `Γ`. (Recall map inclusion, `Γ ⊆ Γ'`, from
225added-- the `Typeclasses` chapter.)
226added
227addedtheorem weakening (Γ Γ' : Context) (t : Tm) (T : Ty)
228added (hi : Γ ⊆ Γ') (hT : <{ ~Γ ⊢ ~t ⦂ ~T }>) : <{ ~Γ' ⊢ ~t ⦂ ~T }> := by
229added induction hT generalizing Γ' with
230added | var _ x _ h => exact .var _ x _ (hi x _ h)
231added | abs _ x _ _ _ _ ih => exact .abs _ x _ _ _ (ih _ (PartialMap.update_subset _ _ _ _ hi))
232added | app _ _ _ _ _ _ _ ih₁ ih₂ => exact .app _ _ _ _ _ (ih₁ _ hi) (ih₂ _ hi)
233added | tru => exact .tru _
234added | fls => exact .fls _
235added | ite _ _ _ _ _ _ _ _ ih₁ ih₂ ih₃ => exact .ite _ _ _ _ _ (ih₁ _ hi) (ih₂ _ hi) (ih₃ _ hi)
236added
237added-- The following simple corollary is what we actually need
238added-- below.
239added
240addedtheorem weakening_empty (Γ : Context) (t : Tm) (T : Ty) (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) :
241added <{ ~Γ ⊢ ~t ⦂ ~T }> :=
242added weakening _ _ _ _
243added (fun x T h => by rw [PartialMap.apply_empty] at h; cases h) hT
244added
245added-- ### The Substitution Lemma
246added
247added-- Now we come to the conceptual heart of the proof that
248added-- reduction preserves types — namely, the observation that
249added-- *substitution* preserves types.
250added
251added-- The *substitution lemma* says:
252added
253added-- - Suppose we have a term `t` with a free variable `x`, and
254added-- suppose we've been able to assign a type `T` to `t` under
255added-- the assumption that `x` has some type `U`.
256added
257added-- - Also, suppose that we have some other term `v` and that
258added-- we've shown that `v` has type `U`.
259added
260added-- - Then we can substitute `v` for each of the occurrences of
261added-- `x` in `t` and obtain a new term that still has type `T`.
262added
263addedtheorem substitution_preserves_typing (Γ : Context) (x : String) (U : Ty)
264added (t v : Tm) (T : Ty)
265added (hT : <{ ~x ↦ ~U ; ~Γ ⊢ ~t ⦂ ~T }>) (hv : <{ ∅ ⊢ ~v ⦂ ~U }>) :
266added <{ ~Γ ⊢ [~x := ~v] ~t ⦂ ~T }> := by
267added -- By induction on `t`; in each case we get at the derivation of `hT`.
268added induction t generalizing Γ T with
269added | var y =>
270added cases hT with
271added | var _ _ _ h =>
272added by_cases hxy : x = y
273added · subst hxy
274added rw [PartialMap.update_eq] at h
275added rw [subst_var_eq]
276added have hUT : U = T := Option.some.inj h
277added subst hUT
278added exact weakening_empty _ _ _ hv
279added · rw [PartialMap.update_neq _ _ _ hxy] at h
280added rw [subst_var_ne _ _ _ hxy]
281added exact .var _ y _ h
282added | app t₁ t₂ ih₁ ih₂ =>
283added cases hT with
284added | app _ _ _ _ _ h₁ h₂ => rw [subst_app]; exact .app _ _ _ _ _ (ih₁ _ _ h₁) (ih₂ _ _ h₂)
285added | abs y S t₁ ih =>
286added cases hT with
287added | abs _ _ _ _ _ h =>
288added by_cases hxy : x = y
289added · subst hxy
290added rw [subst_abs_eq]
291added rw [PartialMap.update_shadow] at h
292added exact .abs _ _ _ _ _ h
293added · rw [subst_abs_ne _ _ _ _ _ hxy]
294added rw [PartialMap.update_permute _ _ _ _ _ (Ne.symm hxy)] at h
295added exact .abs _ _ _ _ _ (ih _ _ h)
296added | tru => cases hT with | tru => rw [subst_tru]; exact .tru _
297added | fls => cases hT with | fls => rw [subst_fls]; exact .fls _
298added | ite c t e ihc iht ihe =>
299added cases hT with
300added | ite _ _ _ _ _ h₁ h₂ h₃ =>
301added rw [subst_ite]
302added exact .ite _ _ _ _ _ (ihc _ _ h₁) (iht _ _ h₂) (ihe _ _ h₃)
303added
304added-- ### Main Theorem
305added
306added-- We now have the ingredients we need to prove preservation:
307added-- if a closed, well-typed term `t` has type `T` and takes a
308added-- step to `t'`, then `t'` is also a closed term with type `T`.
309added-- In other words, the small-step reduction relation preserves
310added-- types.
311added
312addedtheorem preservation (t t' : Tm) (T : Ty)
313added (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) (hs : t ⟶ t') : <{ ∅ ⊢ ~t' ⦂ ~T }> := by
314added generalize hΓ : (∅ : Context) = Γ at hT
315added induction hT generalizing t' with
316added | var => cases hs
317added | abs => cases hs
318added | tru => cases hs
319added | fls => cases hs
320added | app Γ T₁ T₂ t₁ t₂ h₁ h₂ ih₁ ih₂ =>
321added subst hΓ
322added cases hs with
323added | appAbs _ _ _ _ _ =>
324added -- The one interesting case: the desired result is the substitution lemma.
325added cases h₁ with
326added | abs _ _ _ _ _ hb => exact substitution_preserves_typing _ _ _ _ _ _ hb h₂
327added | app1 _ t₁' _ h => exact .app _ _ _ _ _ (ih₁ t₁' h rfl) h₂
328added | app2 _ _ t₂' _ h => exact .app _ _ _ _ _ h₁ (ih₂ t₂' h rfl)
329added | ite Γ t₁ t₂ t₃ T₁ h₁ h₂ h₃ ih₁ ih₂ ih₃ =>
330added subst hΓ
331added cases hs with
332added | ifTrue => exact h₂
333added | ifFalse => exact h₃
334added | ifStep _ t₁' _ _ h => exact .ite _ _ _ _ _ (ih₁ t₁' h rfl) h₂ h₃
335added
336addedend Stlc

338added-- Let's extend the STLC with a base type of numbers, some
339added-- constants, and some primitive operators.
340added
341addednamespace StlcArith
342added
343added-- To types, we add a base type of natural numbers (and remove
344added-- booleans, for brevity).
345added
346addedinductive Ty where
347added | arrow (T₁ T₂ : Ty)
348added | nat
349added
350added-- To terms, we add natural number constants, along with
351added-- successor, predecessor, multiplication, and zero-testing.
352added
353addedinductive Tm where
354added | var (x : String)
355added | app (t₁ t₂ : Tm)
356added | abs (x : String) (T : Ty) (t : Tm)
357added | const (n : Nat)
358added | succ (t : Tm)
359added | pred (t : Tm)
360added | mult (t₁ t₂ : Tm)
361added | ite0 (c t e : Tm)

363added-- _Details:_ Notation encoding: types
364added
365added-- The type grammar needs no new productions: `Nat` is a bare
366added-- identifier, which the template already accepts, and arrows
367added-- and parentheses are unchanged. Only the `macro_rules` are
368added-- new, and they differ from the STLC's in just two places —
369added-- the identifier `Nat` names this language's base type, and
370added-- the arrow builds this language's `StlcArith.Ty.arrow`.
371added
372addedscoped macro_rules (kind := Stlc.tyBracket)
373added | `(<{ ~$T:term }>) => pure T
374added | `(<{ ($T:stlcTy) }>) => `(<{ $T:stlcTy }>)
375added | `(<{ $x:ident }>) =>
376added match x.getId.toString with
377added | "Nat" => `(Ty.nat)
378added | _ => `(($x : Ty))
379added | `(<{ $T₁:stlcTy → $T₂:stlcTy }>) => `(Ty.arrow <{ $T₁:stlcTy }> <{ $T₂:stlcTy }>)
380added | `(<{ $T₁:stlcTy -> $T₂:stlcTy }>) => `(Ty.arrow <{ $T₁:stlcTy }> <{ $T₂:stlcTy }>)
381added
382added-- _Details:_ Notation encoding: terms
383added
384added-- Terms do need new productions: a numeral, an infix `*`, and
385added-- the zero test. Multiplication binds looser than application
386added-- and tighter than `λ`, so `x * y z` multiplies `x` by the
387added-- application `y z`; it associates to the right, so
388added-- `x * y * z` is `x * (y * z)`.
389added
390added-- `succ` and `pred` get no production of their own. Making
391added-- them keywords would reserve those words globally — and we
392added-- would then be unable to write `succ` as a case name in a
393added-- proof, including for Lean's own `Nat`. Instead they are
394added-- written as though they were functions applied to an
395added-- argument, `succ t`, and the application rule below
396added-- recognizes them. `if0` **is** a keyword, since `then` and
397added-- `else` leave no other option; that is why the constructor
398added-- above is called `StlcArith.Tm.ite0` rather than `if0`, just
399added-- as the STLC's conditional is `Stlc.Tm.ite`.
400added
401addedscoped syntax:max num : stlcTm
402addedscoped syntax:60 stlcTm:61 " * " stlcTm:60 : stlcTm
403addedscoped syntax:50 "if0 " stlcTm:51 " then " stlcTm:50 " else " stlcTm:50 : stlcTm
404added
405addedopen Lean in
406addedscoped macro_rules (kind := Stlc.tmBracket)
407added | `(<{ ~$e:term }>) => pure e
408added | `(<{ ($t:stlcTm) }>) => `(<{ $t:stlcTm }>)
409added | `(<{ $n:num }>) => `(Tm.const $n)
410added | `(<{ $x:ident }>) =>
411added match x.getId.toString with
412added | "Nat" => Macro.throwErrorAt x "`Nat` is a type, not a term"
413added | "succ" => Macro.throwErrorAt x "`succ` must be applied to an argument"
414added | "pred" => Macro.throwErrorAt x "`pred` must be applied to an argument"
415added | _ => `(Tm.var $(quote x.getId.toString))
416added | `(<{ $t₁:stlcTm $t₂:stlcTm }>) =>
417added match t₁ with
418added | `(stlcTm| $f:ident) =>
419added match f.getId.toString with
420added | "succ" => `(Tm.succ <{ $t₂:stlcTm }>)
421added | "pred" => `(Tm.pred <{ $t₂:stlcTm }>)
422added | _ => `(Tm.app <{ $t₁:stlcTm }> <{ $t₂:stlcTm }>)
423added | _ => `(Tm.app <{ $t₁:stlcTm }> <{ $t₂:stlcTm }>)
424added | `(<{ λ $x : $T . $t }>) => do
425added `(Tm.abs $(← Stlc.varStr x) <{ $T:stlcTy }> <{ $t:stlcTm }>)
426added | `(<{ $t₁:stlcTm * $t₂:stlcTm }>) => `(Tm.mult <{ $t₁:stlcTm }> <{ $t₂:stlcTm }>)
427added | `(<{ if0 $c then $t else $e }>) =>
428added `(Tm.ite0 <{ $c:stlcTm }> <{ $t:stlcTm }> <{ $e:stlcTm }>)

431added
432added-- As in the Stlc chapter, a delaborator runs the grammar
433added-- backwards, so that goals mentioning these terms and types
434added-- read in the concrete syntax. The parenthesizers registered
435added-- there are for the whole syntax category, so they serve this
436added-- language too and are not repeated.

440addedpartial def delabTyInner : DelabM (TSyntax `stlcTy) := do
441added let stx ←
442added match_expr ← getExpr with
443added | Ty.nat => `(stlcTy| $(mkIdent `Nat):ident)
444added | Ty.arrow _ _ => do
445added let a ← withAppFn <| withAppArg delabTyInner
446added let b ← withAppArg delabTyInner
447added `(stlcTy| $a → $b)
448added | _ => do
449added match ← delab with
450added | `($i:ident) => `(stlcTy| $i:ident)
451added | e => `(stlcTy| ~$e)
452added (⟨·⟩) <$> annotateTermInfo ⟨stx.raw⟩

454addedopen Lean PrettyPrinter Delaborator SubExpr in
455added/-- Rebuild `stlcTm` concrete syntax from a `Tm` value. -/
456addedpartial def delabTmInner : DelabM (TSyntax `stlcTm) := do
457added let stx ←
458added match_expr ← getExpr with
459added | Tm.var _ => do
460added match ← withAppArg delab with

465added | _ => `(stlcTm| ~($(⟨← delab⟩)))
466added | Tm.const _ => do

468added | `($n:num) => `(stlcTm| $n:num)
469added | _ => `(stlcTm| ~($(⟨← delab⟩)))
470added | Tm.app _ _ => do

474added | Tm.abs _ _ _ => do

482added | Tm.pred _ => do
483added let t ← withAppArg delabTmInner
484added `(stlcTm| $(mkIdent `pred):ident $t)
485added | Tm.mult _ _ => do
486added let a ← withAppFn <| withAppArg delabTmInner
487added let b ← withAppArg delabTmInner
488added `(stlcTm| $a * $b)
489added | Tm.ite0 _ _ _ => do
490added let c ← withAppFn <| withAppFn <| withAppArg delabTmInner
491added let t ← withAppFn <| withAppArg delabTmInner
492added let e ← withAppArg delabTmInner
493added `(stlcTm| if0 $c then $t else $e)
