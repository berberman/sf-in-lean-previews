735+--  ### The Constant-Folding Transformation
736+
737+--  An expression is *constant* if it contains no variable references.
738+--
739+--  Constant folding is an optimization that finds constant expressions and
740+--  replaces them by their values.
741+
742+def Aexp.foldConstants (a : Aexp) : Aexp :=
743+  match a with
744+  | .num n => .num n
745+  | .id x => .id x
746+  | aexp { ~a₁ + ~a₂ } =>
747+    match a₁.foldConstants, a₂.foldConstants with
748+    | .num n₁, .num n₂ => .num (n₁ + n₂)
749+    | a₁', a₂' => aexp { ~a₁' + ~a₂' }
750+  | aexp { ~a₁ - ~a₂ } =>
751+    match a₁.foldConstants, a₂.foldConstants with
752+    | .num n₁, .num n₂ => .num (n₁ - n₂)
753+    | a₁', a₂' => aexp { ~a₁' - ~a₂' }
754+  | aexp { ~a₁ * ~a₂ } =>
755+    match a₁.foldConstants, a₂.foldConstants with
756+    | .num n₁, .num n₂ => .num (n₁ * n₂)
757+    | a₁', a₂' => aexp { ~a₁' * ~a₂' }
758+
759+@[simp]
760+theorem Aexp.foldConstants_num (n : Nat) : (Aexp.num n).foldConstants = .num n := rfl
761+@[simp]
762+theorem Aexp.foldConstants_id (x : Ident) : (Aexp.id x).foldConstants = .id x := rfl

764+theorem Aexp.foldConstants_cases (a₁ a₂ : Aexp) :
765+    (∃ n₁ n₂, a₁.foldConstants = .num n₁ ∧ a₂.foldConstants = .num n₂) ∨
766+    (aexp {~a₁ + ~a₂}).foldConstants = (aexp {~a₁.foldConstants + ~a₂.foldConstants}) ∧
767+    (aexp {~a₁ - ~a₂}).foldConstants = (aexp {~a₁.foldConstants - ~a₂.foldConstants}) ∧
768+    (aexp {~a₁ * ~a₂}).foldConstants = (aexp {~a₁.foldConstants * ~a₂.foldConstants}) := by

801+def Bexp.foldConstants (b : Bexp) : Bexp :=
802+  match b with
803+  | bexp { true } => bexp { true }
804+  | bexp { false } => bexp { false }
805+  | bexp { ~a₁ = ~a₂ } =>
806+    match a₁.foldConstants, a₂.foldConstants with
807+    | .num n₁, .num n₂ => if n₁ = n₂ then bexp { true } else bexp {false}
808+    | a₁', a₂' => bexp { ~a₁' = ~a₂' }
809+  | bexp { ~a₁ ≠ ~a₂ } =>

811+    | .num n₁, .num n₂ => if n₁ ≠ n₂ then bexp { true } else bexp {false}
812+    | a₁', a₂' => bexp { ~a₁' ≠ ~a₂' }
813+  | bexp { ~a₁ ≤ ~a₂ } =>
814+    match a₁.foldConstants, a₂.foldConstants with
815+    | .num n₁, .num n₂ => if n₁ ≤ n₂ then bexp { true } else bexp {false}
816+    | a₁', a₂' => bexp { ~a₁' ≤ ~a₂' }
817+  | bexp { ~a₁ > ~a₂ } =>
818+    match a₁.foldConstants, a₂.foldConstants with
819+    | .num n₁, .num n₂ => if n₁ > n₂ then bexp { true } else bexp {false}
820+    | a₁', a₂' => bexp { ~a₁' > ~a₂' }
821+  | bexp { ¬ ~b₁ } =>

823+    | bexp { true } => bexp { false }
