708+--  ## Program Transformation
709+
710+--  A *program transformation* is a function that takes a program as input
711+--  and produces a modified program as output. Compiler optimizations such
712+--  as constant folding are canonical examples, but there are many others.
713+
714+def Aexp.trans_sound (trans : Aexp → Aexp) : Prop :=
715+  ∀ (a : Aexp),
716+    a.Equiv (trans a)
717+
718+def Bexp.trans_sound (trans : Bexp → Bexp) : Prop :=
719+  ∀ (b : Bexp),
720+    b.Equiv (trans b)
721+
722+def Com.trans_sound (trans : Com → Com) : Prop :=
723+  ∀ (c : Com),
724+    c.Equiv (trans c)

726+--  ### The Constant-Folding Transformation
727+
728+--  An expression is *constant* if it contains no variable references.
729+
730+--  Constant folding is an optimization that finds constant expressions and
731+--  replaces them by their values.
732+
733+def Aexp.fold_constants (a: Aexp) : Aexp :=
734+  match a with
735+  | .num n => .num n
736+  | .id x => .id x
737+  | aexp { ~a1 + ~a2 } => 
738+    match a1.fold_constants, a2.fold_constants with
739+    | .num n1, .num n2 => .num (n1 + n2)
740+    | a1', a2' => aexp { ~a1' + ~a2'}
741+  | aexp { ~a1 - ~a2 } => 
742+    match a1.fold_constants, a2.fold_constants with
743+    | .num n1, .num n2 => .num (n1 - n2)
744+    | a1', a2' => aexp { ~a1' - ~a2'}
745+  | aexp { ~a1 * ~a2 } => 
746+    match a1.fold_constants, a2.fold_constants with
747+    | .num n1, .num n2 => .num (n1 * n2)
748+    | a1', a2' => aexp { ~a1' * ~a2'}
749+
750+example : (aexp { (1 + 2) * X}).fold_constants = (aexp { 3 * X }) := by rfl
751+
752+--  Note that this version of constant folding doesn't do other "obvious"
753+--  things like eliminating trivial additions (e.g., rewriting `0 + X` to
754+--  just `X`).: we are focusing on a single optimization for the sake of
755+--  simplicity.
756+
757+--  It is not hard to incorporate other ways of simplifying expressions --
758+--  the definitions and proofs just get longer. We'll consider some in the
759+--  exercises.
760+
761+example : (aexp { X - ((0 * 6) + Y) }).fold_constants = (aexp { X - (0 + Y) }) := by rfl
762+
763+--  Not only can we lift `Aexp.fold_constants` to `Bexp` in the `Bexp.eq`,
764+--  `Bexp.neq`, and `Bexp.le` cases, we can also look for constant
765+--  *boolean* expressions and evaluate them in place as well.
766+
767+def Bexp.fold_constants (b : Bexp) : Bexp :=
768+  match b with
769+  | bexp { true } => bexp { true }
770+  | bexp { false } => bexp { false }
771+  | bexp { ~a1 = ~a2 } => 
772+    match a1.fold_constants, a2.fold_constants with
773+    | .num n1, .num n2 => if n1 = n2 then bexp { true } else bexp {false}
774+    | a1', a2' => bexp { ~a1' = ~a2' }
775+  | bexp { ~a1 ≠ ~a2 } => 
776+    match a1.fold_constants, a2.fold_constants with
777+    | .num n1, .num n2 => if n1 ≠ n2 then bexp { true } else bexp {false}
778+    | a1', a2' => bexp { ~a1' ≠ ~a2' }
779+  | bexp { ~a1 ≤ ~a2 } => 
780+    match a1.fold_constants, a2.fold_constants with
781+    | .num n1, .num n2 => if n1 ≤ n2 then bexp { true } else bexp {false}
782+    | a1', a2' => bexp { ~a1' ≤ ~a2' }
783+  | bexp { ~a1 > ~a2 } => 
784+    match a1.fold_constants, a2.fold_constants with
785+    | .num n1, .num n2 => if n1 > n2 then bexp { true } else bexp {false}
786+    | a1', a2' => bexp { ~a1' > ~a2' }
787+  | bexp { ¬~b1 } =>
788+    match b1.fold_constants with
789+    | bexp { true } => bexp { false }
790+    | bexp { false } => bexp { true }
791+    | b1' => bexp { ¬~b1' }
792+  | bexp { ~b1 ∧ ~b2 } =>
793+    match b1.fold_constants, b2.fold_constants with
794+    | bexp { true }, bexp { true } => bexp { true }
795+    | bexp { true }, bexp { false } => bexp { false }
796+    | bexp { false }, bexp { true } => bexp { false }
797+    | bexp { false }, bexp { false } => bexp { false }
798+    | b1', b2' => bexp { ~b1' ∧ ~b2' }
799+
800+example : (bexp { true ∧ ¬( false ∧ true) }).fold_constants = (bexp { true }) := by rfl
801+example : (bexp { (X = Y) ∧ ( 0 = (2 - (1 + 1))) }).fold_constants = (bexp { (X = Y) ∧ true }) := by rfl

803+--  To fold constants in a command, we simply apply the appropriate folding
804+--  functions on all embedded expressions.
805+
806+def Com.fold_constants (c : Com) : Com :=
807+  match c with
808+  | imp { skip } => imp { skip }
809+  | imp { x := ~a } => imp { x := ~a.fold_constants }
810+  | imp { ~c1 ; ~c2 } =>  imp { ~c1.fold_constants ; ~c2.fold_constants }
811+  | imp { if (~b) {~c1} else {~c2}} =>
812+    match b.fold_constants with
813+    | bexp { true } => c1.fold_constants
814+    | bexp { false } => c2.fold_constants
815+    | b' => imp { if (~b') {~c1.fold_constants} else {~c2.fold_constants}}
816+  | imp { while (~b) {~c}} =>
817+    match b.fold_constants with
818+    | bexp { true } => imp { while (true) { skip }}
819+    | bexp { false } => imp { skip }
820+    | b' => imp { while (~b') {~c.fold_constants}}
