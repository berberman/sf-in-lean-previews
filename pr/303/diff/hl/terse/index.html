314+def Aexp.fold_constants (a: Aexp) : Aexp :=
315+  match a with
316+  | .num n => .num n
317+  | .id x => .id x
318+  | aexp { ~a1 + ~a2 } => 
319+    match a1.fold_constants, a2.fold_constants with
320+    | .num n1, .num n2 => .num (n1 + n2)
321+    | a1', a2' => aexp { ~a1' + ~a2'}
322+  | aexp { ~a1 - ~a2 } => 
323+    match a1.fold_constants, a2.fold_constants with
324+    | .num n1, .num n2 => .num (n1 - n2)
325+    | a1', a2' => aexp { ~a1' - ~a2'}
326+  | aexp { ~a1 * ~a2 } => 
327+    match a1.fold_constants, a2.fold_constants with
328+    | .num n1, .num n2 => .num (n1 * n2)
329+    | a1', a2' => aexp { ~a1' * ~a2'}
330+
331+example : (aexp { (1 + 2) * X}).fold_constants = (aexp { 3 * X }) := by rfl
332+
333+example : (aexp { X - ((0 * 6) + Y) }).fold_constants = (aexp { X - (0 + Y) }) := by rfl

335+def Bexp.fold_constants (b : Bexp) : Bexp :=
336+  match b with
337+  | bexp { true } => bexp { true }
338+  | bexp { false } => bexp { false }
339+  | bexp { ~a1 = ~a2 } => 
340+    match a1.fold_constants, a2.fold_constants with
341+    | .num n1, .num n2 => if n1 = n2 then bexp { true } else bexp {false}
342+    | a1', a2' => bexp { ~a1' = ~a2' }
343+  | bexp { ~a1 ≠ ~a2 } => 
344+    match a1.fold_constants, a2.fold_constants with
345+    | .num n1, .num n2 => if n1 ≠ n2 then bexp { true } else bexp {false}
346+    | a1', a2' => bexp { ~a1' ≠ ~a2' }
347+  | bexp { ~a1 ≤ ~a2 } => 
348+    match a1.fold_constants, a2.fold_constants with
349+    | .num n1, .num n2 => if n1 ≤ n2 then bexp { true } else bexp {false}
350+    | a1', a2' => bexp { ~a1' ≤ ~a2' }
351+  | bexp { ~a1 > ~a2 } => 
352+    match a1.fold_constants, a2.fold_constants with
353+    | .num n1, .num n2 => if n1 > n2 then bexp { true } else bexp {false}
354+    | a1', a2' => bexp { ~a1' > ~a2' }
355+  | bexp { ¬~b1 } =>
356+    match b1.fold_constants with
357+    | bexp { true } => bexp { false }
358+    | bexp { false } => bexp { true }
359+    | b1' => bexp { ¬~b1' }
360+  | bexp { ~b1 ∧ ~b2 } =>
361+    match b1.fold_constants, b2.fold_constants with
362+    | bexp { true }, bexp { true } => bexp { true }
363+    | bexp { true }, bexp { false } => bexp { false }
364+    | bexp { false }, bexp { true } => bexp { false }
365+    | bexp { false }, bexp { false } => bexp { false }
366+    | b1', b2' => bexp { ~b1' ∧ ~b2' }
367+
368+example : (bexp { true ∧ ¬( false ∧ true) }).fold_constants = (bexp { true }) := by rfl
369+example : (bexp { (X = Y) ∧ ( 0 = (2 - (1 + 1))) }).fold_constants = (bexp { (X = Y) ∧ true }) := by rfl

371+def Com.fold_constants (c : Com) : Com :=
372+  match c with
373+  | imp { skip } => imp { skip }
374+  | imp { x := ~a } => imp { x := ~a.fold_constants }
375+  | imp { ~c1 ; ~c2 } =>  imp { ~c1.fold_constants ; ~c2.fold_constants }
376+  | imp { if (~b) {~c1} else {~c2}} =>
377+    match b.fold_constants with
378+    | bexp { true } => c1.fold_constants
379+    | bexp { false } => c2.fold_constants
380+    | b' => imp { if (~b') {~c1.fold_constants} else {~c2.fold_constants}}
381+  | imp { while (~b) {~c}} =>
382+    match b.fold_constants with
383+    | bexp { true } => imp { while (true) { skip }}
384+    | bexp { false } => imp { skip }
385+    | b' => imp { while (~b') {~c.fold_constants}}
386+
