330+def Aexp.foldConstants (a : Aexp) : Aexp :=
331+  match a with
332+  | .num n => .num n
333+  | .id x => .id x
334+  | aexp { ~a₁ + ~a₂ } =>
335+    match a₁.foldConstants, a₂.foldConstants with
336+    | .num n₁, .num n₂ => .num (n₁ + n₂)
337+    | a₁', a₂' => aexp { ~a₁' + ~a₂' }
338+  | aexp { ~a₁ - ~a₂ } =>
339+    match a₁.foldConstants, a₂.foldConstants with
340+    | .num n₁, .num n₂ => .num (n₁ - n₂)
341+    | a₁', a₂' => aexp { ~a₁' - ~a₂' }
342+  | aexp { ~a₁ * ~a₂ } =>
343+    match a₁.foldConstants, a₂.foldConstants with
344+    | .num n₁, .num n₂ => .num (n₁ * n₂)
345+    | a₁', a₂' => aexp { ~a₁' * ~a₂' }

352+theorem Aexp.foldConstants_cases (a₁ a₂ : Aexp) :
353+    (∃ n₁ n₂, a₁.foldConstants = .num n₁ ∧ a₂.foldConstants = .num n₂) ∨
354+    (aexp {~a₁ + ~a₂}).foldConstants = (aexp {~a₁.foldConstants + ~a₂.foldConstants}) ∧
355+    (aexp {~a₁ - ~a₂}).foldConstants = (aexp {~a₁.foldConstants - ~a₂.foldConstants}) ∧
356+    (aexp {~a₁ * ~a₂}).foldConstants = (aexp {~a₁.foldConstants * ~a₂.foldConstants}) := by

358+  | num n₁ =>

360+    | num n₂ =>
361+      left
362+      exists n₁, n₂

483+theorem Aexp.foldConstants_sound : TransSound foldConstants := by
