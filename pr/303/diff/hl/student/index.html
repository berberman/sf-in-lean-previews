580+--  ## Program Transformation
581+
582+--  A *program transformation* is a function that takes a program as input
583+--  and produces a modified program as output. Compiler optimizations such
584+--  as constant folding are canonical examples, but there are many others.
585+
586+def Aexp.trans_sound (trans : Aexp → Aexp) : Prop :=
587+  ∀ (a : Aexp),
588+    a.Equiv (trans a)
589+
590+def Bexp.trans_sound (trans : Bexp → Bexp) : Prop :=
591+  ∀ (b : Bexp),
592+    b.Equiv (trans b)
593+
594+def Com.trans_sound (trans : Com → Com) : Prop :=
595+  ∀ (c : Com),
596+    c.Equiv (trans c)
597+
598+--  ### The Constant-Folding Transformation
599+
600+--  An expression is *constant* if it contains no variable references.
601+
602+--  Constant folding is an optimization that finds constant expressions and
603+--  replaces them by their values.
604+
605+def Aexp.fold_constants (a: Aexp) : Aexp :=
606+  match a with
607+  | .num n => .num n
608+  | .id x => .id x
609+  | aexp { ~a1 + ~a2 } => 
610+    match a1.fold_constants, a2.fold_constants with
611+    | .num n1, .num n2 => .num (n1 + n2)
612+    | a1', a2' => aexp { ~a1' + ~a2'}
613+  | aexp { ~a1 - ~a2 } => 
614+    match a1.fold_constants, a2.fold_constants with
615+    | .num n1, .num n2 => .num (n1 - n2)
616+    | a1', a2' => aexp { ~a1' - ~a2'}
617+  | aexp { ~a1 * ~a2 } => 
618+    match a1.fold_constants, a2.fold_constants with
619+    | .num n1, .num n2 => .num (n1 * n2)
620+    | a1', a2' => aexp { ~a1' * ~a2'}
621+
622+example : (aexp { (1 + 2) * X}).fold_constants = (aexp { 3 * X }) := by rfl
623+
624+--  Note that this version of constant folding doesn't do other "obvious"
625+--  things like eliminating trivial additions (e.g., rewriting `0 + X` to
626+--  just `X`).: we are focusing on a single optimization for the sake of
627+--  simplicity.
628+
629+--  It is not hard to incorporate other ways of simplifying expressions --
630+--  the definitions and proofs just get longer. We'll consider some in the
631+--  exercises.
632+
633+example : (aexp { X - ((0 * 6) + Y) }).fold_constants = (aexp { X - (0 + Y) }) := by rfl
634+
635+--  Not only can we lift `Aexp.fold_constants` to `Bexp` in the `Bexp.eq`,
636+--  `Bexp.neq`, and `Bexp.le` cases, we can also look for constant
637+--  *boolean* expressions and evaluate them in place as well.
638+
639+def Bexp.fold_constants (b : Bexp) : Bexp :=
640+  match b with
641+  | bexp { true } => bexp { true }
642+  | bexp { false } => bexp { false }
643+  | bexp { ~a1 = ~a2 } => 
644+    match a1.fold_constants, a2.fold_constants with
645+    | .num n1, .num n2 => if n1 = n2 then bexp { true } else bexp {false}
646+    | a1', a2' => bexp { ~a1' = ~a2' }
647+  | bexp { ~a1 ≠ ~a2 } => 
648+    match a1.fold_constants, a2.fold_constants with
649+    | .num n1, .num n2 => if n1 ≠ n2 then bexp { true } else bexp {false}
650+    | a1', a2' => bexp { ~a1' ≠ ~a2' }
651+  | bexp { ~a1 ≤ ~a2 } => 
652+    match a1.fold_constants, a2.fold_constants with
653+    | .num n1, .num n2 => if n1 ≤ n2 then bexp { true } else bexp {false}
654+    | a1', a2' => bexp { ~a1' ≤ ~a2' }
655+  | bexp { ~a1 > ~a2 } => 
656+    match a1.fold_constants, a2.fold_constants with
657+    | .num n1, .num n2 => if n1 > n2 then bexp { true } else bexp {false}
658+    | a1', a2' => bexp { ~a1' > ~a2' }
659+  | bexp { ¬~b1 } =>
660+    match b1.fold_constants with
661+    | bexp { true } => bexp { false }
662+    | bexp { false } => bexp { true }
663+    | b1' => bexp { ¬~b1' }
664+  | bexp { ~b1 ∧ ~b2 } =>
665+    match b1.fold_constants, b2.fold_constants with
666+    | bexp { true }, bexp { true } => bexp { true }
667+    | bexp { true }, bexp { false } => bexp { false }
668+    | bexp { false }, bexp { true } => bexp { false }
669+    | bexp { false }, bexp { false } => bexp { false }
670+    | b1', b2' => bexp { ~b1' ∧ ~b2' }
671+
672+example : (bexp { true ∧ ¬( false ∧ true) }).fold_constants = (bexp { true }) := by rfl
673+example : (bexp { (X = Y) ∧ ( 0 = (2 - (1 + 1))) }).fold_constants = (bexp { (X = Y) ∧ true }) := by rfl
674+
675+--  To fold constants in a command, we simply apply the appropriate folding
676+--  functions on all embedded expressions.
677+
678+def Com.fold_constants (c : Com) : Com :=
679+  match c with
680+  | imp { skip } => imp { skip }
681+  | imp { x := ~a } => imp { x := ~a.fold_constants }
682+  | imp { ~c1 ; ~c2 } =>  imp { ~c1.fold_constants ; ~c2.fold_constants }
683+  | imp { if (~b) {~c1} else {~c2}} =>
684+    match b.fold_constants with
685+    | bexp { true } => c1.fold_constants
686+    | bexp { false } => c2.fold_constants
687+    | b' => imp { if (~b') {~c1.fold_constants} else {~c2.fold_constants}}
688+  | imp { while (~b) {~c}} =>
689+    match b.fold_constants with
690+    | bexp { true } => imp { while (true) { skip }}
691+    | bexp { false } => imp { skip }
692+    | b' => imp { while (~b') {~c.fold_constants}}
693+
694+example :
695+  (imp {
696+    X := 4 + 5;
697+    Y := X - 3;
698+    if ((X - Y) = (2 + 4)) {skip} else {Y := 0};
699+    if (0 ≤ (4 - (2 - 1))) {Y := 0} else {skip};
700+    while (Y = 0) {X := X+1}
701+  }).fold_constants =
