584+--  and produces a modified program as output. Compiler optimizations such

605+--  ### The Constant-Folding Transformation
606+
607+--  An expression is *constant* if it contains no variable references.
608+--
609+--  Constant folding is an optimization that finds constant expressions and
610+--  replaces them by their values.
611+
612+def Aexp.foldConstants (a : Aexp) : Aexp :=
613+  match a with
614+  | .num n => .num n
615+  | .id x => .id x
616+  | aexp { ~a₁ + ~a₂ } =>
617+    match a₁.foldConstants, a₂.foldConstants with
618+    | .num n₁, .num n₂ => .num (n₁ + n₂)
619+    | a₁', a₂' => aexp { ~a₁' + ~a₂' }
620+  | aexp { ~a₁ - ~a₂ } =>
621+    match a₁.foldConstants, a₂.foldConstants with
622+    | .num n₁, .num n₂ => .num (n₁ - n₂)
623+    | a₁', a₂' => aexp { ~a₁' - ~a₂' }
624+  | aexp { ~a₁ * ~a₂ } =>
625+    match a₁.foldConstants, a₂.foldConstants with
626+    | .num n₁, .num n₂ => .num (n₁ * n₂)
627+    | a₁', a₂' => aexp { ~a₁' * ~a₂' }
628+
629+@[simp]
630+theorem Aexp.foldConstants_num (n : Nat) : (Aexp.num n).foldConstants = .num n := rfl
631+@[simp]
632+theorem Aexp.foldConstants_id (x : Ident) : (Aexp.id x).foldConstants = .id x := rfl

634+theorem Aexp.foldConstants_cases (a₁ a₂ : Aexp) :
635+    (∃ n₁ n₂, a₁.foldConstants = .num n₁ ∧ a₂.foldConstants = .num n₂) ∨
636+    (aexp {~a₁ + ~a₂}).foldConstants = (aexp {~a₁.foldConstants + ~a₂.foldConstants}) ∧
637+    (aexp {~a₁ - ~a₂}).foldConstants = (aexp {~a₁.foldConstants - ~a₂.foldConstants}) ∧
638+    (aexp {~a₁ * ~a₂}).foldConstants = (aexp {~a₁.foldConstants * ~a₂.foldConstants}) := by

640+  | num n₁ =>

642+    | num n₂ =>

644+      exists n₁, n₂

667+def Bexp.foldConstants (b : Bexp) : Bexp :=
668+  match b with
669+  | bexp { true } => bexp { true }
670+  | bexp { false } => bexp { false }
671+  | bexp { ~a₁ = ~a₂ } =>

673+    | .num n₁, .num n₂ => if n₁ = n₂ then bexp { true } else bexp {false}
674+    | a₁', a₂' => bexp { ~a₁' = ~a₂' }

677+    | .num n₁, .num n₂ => if n₁ ≠ n₂ then bexp { true } else bexp {false}

679+  | bexp { ~a₁ ≤ ~a₂ } =>

681+    | .num n₁, .num n₂ => if n₁ ≤ n₂ then bexp { true } else bexp {false}
682+    | a₁', a₂' => bexp { ~a₁' ≤ ~a₂' }
683+  | bexp { ~a₁ > ~a₂ } =>
684+    match a₁.foldConstants, a₂.foldConstants with
685+    | .num n₁, .num n₂ => if n₁ > n₂ then bexp { true } else bexp {false}
686+    | a₁', a₂' => bexp { ~a₁' > ~a₂' }

744+--  To fold constants in a command, we simply apply the appropriate folding
745+--  functions on all embedded expressions.

747+def Com.foldConstants (c : Com) : Com :=
748+  match c with
749+  | imp { skip } => imp { skip }
750+  | imp { x := ~a } => imp { x := ~a.foldConstants }
751+  | imp { ~c₁ ; ~c₂ } =>  imp { ~c₁.foldConstants ; ~c₂.foldConstants }
752+  | imp { if (~b) {~c₁} else {~c₂}} =>
753+    match b.foldConstants with
754+    | bexp { true } => c₁.foldConstants
755+    | bexp { false } => c₂.foldConstants
756+    | b' => imp { if (~b') {~c₁.foldConstants} else {~c₂.foldConstants}}
757+  | imp { while (~b) {~c}} =>
758+    match b.foldConstants with
