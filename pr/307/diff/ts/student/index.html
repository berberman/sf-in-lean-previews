22+/--
23+  `inversion h` behaves like `cases h` except that where `cases` would fail with
24+  > "Dependent elimination failed: Failed to solve equation ...",
25+
26+  `inversion` instead leaves behind those unsolveable equations in the context.
27+  These equations arise from generalizing the indices of `h`'s type.
28+  In such cases, `h` itself is generalized with a heterogenenous equality `heq`
29+  to the corresponding constructor in each branch.
30+
31+  For instance, given a hypothesis `h : f n ≤ 0`,
32+  `inversion h` will produce one goal for the case `Nat.le.refl {n} : n ≤ n`
33+  with the equations `eq : f n = 0` and `heq : h ≍ Nat.le.refl`,
34+  rather than failing to solve `f n = 0`.
35+
36+  * If `inversion +clear h` is set, then `heq` will be `clear`ed from the context if present,
37+    along with `h` if there are no dependencies on it.
38+    This is the usual behavior of `cases`.
39+  * The form `inversion t with (tac ...) | tag₁ x ... => tac ... | ... | tagₙ z ... => tac ...`
40+    is supported, similar to that of `cases` and `induction`.

60+/--
61+  This function is similar to `forallMetaTelescopeReducing`: Given `e` of the
62+  form `forall ..xs, A`, this combinator will create a new metavariable for
63+  each `x` in `xs` until it reaches an `x` whose type is defeq to `t`,
64+  and instantiate `A` with these, while also reducing `A` if needed.
65+  It uses `forallMetaTelescopeReducing`.
66+
67+  This function returns a triple `(mvs, bis, out)` where
68+  - `mvs` is an array containing the new metavariables.
69+  - `bis` is an array containing the binder infos for the `mvs`.
70+  - `out` is `e` but instantiated with the `mvs`.
71+
72+  Lifted from [https://github.com/leanprover-community/mathlib4/blob/master/Mathlib/Lean/Meta/Basic.lean#L41].

74+def forallMetaTelescopeReducingUntilDefEq
75+    (e t : Expr) (kind : MetavarKind := MetavarKind.natural) :
76+    MetaM (Array Expr × Array BinderInfo × Expr) := do
77+  let (ms, bs, tp) ← forallMetaTelescopeReducing e (some 1) kind
78+  unless ms.size == 1 do
79+    if ms.size == 0 then throwError m!"Failed: {← ppExpr e} is not the type of a function."
80+    else throwError m!"Failed"
81+  let mut mvs := ms
82+  let mut bis := bs
83+  let mut out : Expr := tp
84+  while !(← isDefEq (← inferType mvs.toList.getLast!) t) do
85+    let (ms, bs, tp) ← forallMetaTelescopeReducing out (some 1) kind
86+    unless ms.size == 1 do
87+      throwError m!"Failed to find {← ppExpr t} as the type of a parameter of {← ppExpr e}."
88+    mvs := mvs ++ ms
89+    bis := bis ++ bs
90+    out := tp
91+  return (mvs, bis, out)

93+/-- Given equations `eqs` in the local context of `mvarId`,
94+  unify them using [Lean.Meta.unifyEq?],
95+  but if unification fails, leave the equation in the context.
96+  This means we should never get a "Dependent elimination failed" error.
97+  While marked partial, this _should_ always terminate,
98+  decreasing on the sum of the sizes of the terms of each equation.
99+
100+  Lifted from [Lean.Meta.Cases.unifyEqs]. -/
101+partial def unifyEqs (eqs : List FVarId) (mvarId : MVarId) (subst : FVarSubst) (caseName? : Option Name) : MetaM (MVarId × FVarSubst) := withIncRecDepth do
102+  match eqs with
103+  | [] => return (mvarId, subst)
104+  | eq :: eqs =>
105+    let some { mvarId, subst, numNewEqs } ← Option.join <$> (observing? $ unifyEq? mvarId eq subst MVarId.acyclic caseName?)
106+      | unifyEqs eqs mvarId subst caseName?
107+    let (newEqs, mvarId) ← mvarId.introN numNewEqs
108+    let eqs := eqs.map (Expr.fvarId! ∘ subst.get)
109+    unifyEqs (newEqs.toList ++ eqs) mvarId subst caseName?

111+/-- Lifted from [Lean.Meta.Cases.unifyCasesEqs]. -/
112+def unifyCasesEqs (eqs : List FVarId) (subgoals : Array CasesSubgoal) : MetaM (Array CasesSubgoal) :=
113+  subgoals.filterMapM fun s => do
114+    let eqs := eqs.filterMap (Expr.fvarId? ∘ s.subst.get)
115+    let (mvarId, subst) ← unifyEqs eqs s.mvarId s.subst s.ctorName
116+    if (← mvarId.isAssignedOrDelayedAssigned) then return none
117+    else return some { s with
118+      mvarId := mvarId,
119+      subst  := subst,
120+      fields := s.fields.map (subst.apply ·)
121+    }

172+/-- Perform inversion on the given variable in the context of the given goal.
173+  Behaves like [Lean.Meta.Cases.cases] and generalizes indices if they aren't independent,
174+  except instead of calling [Lean.Meta.Cases.unifyEqs], which may fail
175+  if the generalizing equations can't all be destructed and substituted away,
176+  it calls our [unifyCasesEqs] above, which keeps unsolvable equations in the context. -/
177+def _root_.Lean.MVarId.inversion (mvarId : MVarId) (h : FVarId) (config : InversionConfig)
178+    (givenNames : Array AltVarNames := #[]) (interestingCtors? : Option (Array Name) := none) :
179+    TacticM (Array CasesSubgoal) := mvarId.withContext do
180+  let goal ← getMainGoal
181+  goal.withContext do
182+    let some ctx ← mkCasesContext? h
183+    | throwTacticEx `inversion goal "Target {Expr.fvar h} does not belong to an inductive type"
184+    -- If the target's type isn't a predicate with indices, behave like `cases`
185+    if ← hasIndepIndices ctx then
186+      let subgoals ← inductionCasesOn goal h givenNames ctx (interestingCtors? := interestingCtors?)
187+      let subgoals ← if config.clear
188+        then casesClearMany subgoals #[h]
189+        else pure subgoals
190+      return subgoals
191+    -- Otherwise, use custom [unifyEqs] to not fail on unifying HEqs
192+    else
193+      let gis@⟨newGoal, _, newTarget, numEqs⟩ ← generalizeIndices goal h
194+      let (newEqs, newGoal) ← newGoal.introN numEqs
195+      let some targetEq := newEqs.back?
196+      | throwTacticEx `inversion newGoal "Failed to generalize target"
197+      let subgoals ← inductionCasesOn newGoal newTarget givenNames ctx (interestingCtors? := interestingCtors?)
198+      let subgoals ← elimAuxIndices gis subgoals
199+      let subgoals ← unifyCasesEqs newEqs.toList subgoals
200+      let subgoals ← if config.clear
201+        then casesClearMany subgoals #[h, targetEq]
202+        else pure subgoals
203+      return subgoals

205+/-- Given an inversion alternative and a list of goals,
206+  solve the tagged goal with the provided tactics,
207+  throwing an error if the goal cannot be found or solved. -/
208+def evalInvAlt (goals : List MVarId) (alt : TSyntax `invAlt) : TacticM (List MVarId) :=
209+  match alt with
210+  | `(invAlt| | $tag:ident $vars:binderIdent* => $tactics:tacticSeq) => do
211+    if let some goal ← goals.findM? (matchTag tag.getId <$> ·.getUserName) then
212+      return [← trySolveGoal vars tactics goal]
213+    else throwError m!"Invalid alternative name `{tag.getId}`: {← errorMsg}"
214+  | `(invAlt| | _ $vars:binderIdent* => $tactics:tacticSeq) => do
215+    if !goals.isEmpty then
216+      goals.mapM (trySolveGoal vars tactics)
217+    else throwErrorAt alt m!"Invalid wildcard alternative: {← errorMsg}"
218+  | _ => throwErrorAt alt "Could not parse inversion alternative"
219+where
220+  matchTag tag name : Bool :=
221+    tag == name || tag.isPrefixOf name || tag.isSuffixOf name
222+  trySolveGoal vars tactics goal : TacticM MVarId := do
223+    let goals ← renameInaccessibles goal vars >>= evalTacticAt tactics
224+    unless goals.isEmpty do
225+      reportUnsolvedGoals goals
226+    return goal
227+  errorMsg : TacticM MessageData := do
228+    let goalMsgs ← goals.mapM (.ofName <$> ·.getUserName)

274+  requires `generalizing e : f n = m`, since Lean does not perform

29+theorem canonical_forms_bool (t : Tm) (hT : <{ ∅ ⊢ ~t ⦂ Bool }>) (hv : t.IsValue) :
30+    t = <{ true }> ∨ t = <{ false }> := by
31+  cases hv with
32+  | abs x T t₁ => cases hT
33+  | tru => left; rfl
34+  | fls => right; rfl

38+    ∃ x u, t = <{ λ ~x : ~T₁ . ~u }> := by

45+--  ## Progress
46+
47+--  The *progress* theorem tells us that closed, well-typed terms are not
48+--  stuck: either a well-typed term is a value, or it can take a reduction
49+--  step. The proof is a relatively straightforward extension of the
50+--  progress proof we saw in the Types chapter. We give the proof in
51+--  English first, then the formal version.
52+
53+--  *Proof*: By induction on the derivation of `⊢ t ⦂ T`.
54+
55+--  - The last rule of the derivation cannot be `HasType.var`, since a
56+--    variable is never well typed in an empty context.
57+
58+--  - The `HasType.tru`, `HasType.fls`, and `HasType.abs` cases are
59+--    trivial, since in each of these cases we can see by inspecting the
60+--    rule that `t` is a value.
61+
62+--  - If the last rule of the derivation is `HasType.app`, then `t` has the
63+--    form `t₁ t₂` for some `t₁` and `t₂`, where `⊢ t₁ ⦂ T₂ → T` and
64+--    `⊢ t₂ ⦂ T₂` for some type `T₂`. The induction hypothesis for the
65+--    first subderivation says that either `t₁` is a value or else it can
66+--    take a reduction step.
67+
68+--    - If `t₁` is a value, then consider `t₂`, which by the induction
69+--      hypothesis for the second subderivation must also either be a value
70+--      or take a step.
71+
72+--      - Suppose `t₂` is a value. Since `t₁` is a value with an arrow
73+--        type, it must be a lambda abstraction; hence `t₁ t₂` can take a
74+--        step by `Step.appAbs`.
75+
76+--      - Otherwise, `t₂` can take a step, and hence so can `t₁ t₂` by
77+--        `Step.app2`.
78+
79+--    - If `t₁` can take a step, then so can `t₁ t₂` by `Step.app1`.
80+
81+--  - If the last rule of the derivation is `HasType.ite`, then
82+--    `t = if t₁ then t₂ else t₃`, where `t₁` has type `Bool`. The first IH
83+--    says that `t₁` either is a value or takes a step.
84+
85+--    - If `t₁` is a value, then since it has type `Bool` it must be either
86+--      `true` or `false`. If it is `true`, then `t` steps to `t₂`;
87+--      otherwise it steps to `t₃`.
88+
89+--    - Otherwise, `t₁` takes a step, and therefore so does `t` (by
90+--      `Step.ifStep`).
91+
92+theorem progress (t : Tm) (T : Ty) (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) :
93+    t.IsValue ∨ ∃ t', t ⟶ t' := by
94+  generalize hΓ : (∅ : Context) = Γ at hT
95+  induction hT with
96+  | var Γ x T₁ h =>
97+    subst hΓ
98+    -- Contradictory: variables cannot be typed in an empty context.
99+    rw [PartialMap.getElem_empty] at h
100+    cases h
101+  | abs => left; constructor
102+  | tru => left; constructor
103+  | fls => left; constructor
104+  | app Γ T₁ T₂ t₁ t₂ h₁ h₂ ih₁ ih₂ =>
105+    -- `t = t₁ t₂`.  Proceed by cases on whether `t₁` is a value or steps.
106+    right
107+    cases ih₁ hΓ with
108+    | inl hv₁ =>
109+      cases ih₂ hΓ with
110+      | inl hv₂ =>
111+        obtain ⟨x, u, rfl⟩ := canonical_forms_fun t₁ _ _ (hΓ ▸ h₁) hv₁
112+        exists <{ [~x := ~t₂] ~u }>
113+        constructor
114+        assumption
115+      | inr hs₂ =>
116+        obtain ⟨t₂', h⟩ := hs₂
117+        exists <{ ~t₁ ~t₂' }>
118+        constructor <;> assumption
119+    | inr hs₁ =>
120+      obtain ⟨t₁', h⟩ := hs₁
121+      exists <{ ~t₁' ~t₂ }>
122+      constructor <;> assumption
123+  | ite Γ t₁ t₂ t₃ T₁ h₁ h₂ h₃ ih₁ ih₂ ih₃ =>
124+    right
125+    cases ih₁ hΓ with
126+    | inl hv₁ =>
127+      cases canonical_forms_bool t₁ (hΓ ▸ h₁) hv₁ with
128+      | inl he =>
129+        subst he
130+        exists t₂
131+        constructor
132+      | inr he =>
133+        subst he
134+        exists t₃
135+        constructor
136+    | inr hs₁ =>
137+      obtain ⟨t₁', h⟩ := hs₁
138+      exists <{ if ~t₁' then ~t₂ else ~t₃ }>
139+      constructor
140+      assumption

152+--  ## Preservation
153+
154+--  The other half of the type soundness property is the preservation of
155+--  types during reduction. For this part, we'll need to develop some
156+--  technical machinery for reasoning about variables and substitution.
157+--  Working from top to bottom (from the high-level property we are
158+--  actually interested in to the lowest-level technical lemmas that are
159+--  needed by various cases of the more interesting proofs), the story goes
160+--  like this:
161+
162+--  - The *preservation theorem* is proved by induction on a typing
163+--    derivation and case analysis on the step relation, pretty much as we
164+--    did in the Types chapter. The one case that is significantly
165+--    different is the one for the `Step.appAbs` rule, whose definition
166+--    uses the substitution operation. To see that this step preserves
167+--    typing, we need to know that the substitution itself does. So we
168+--    prove a...
169+
170+--  - *substitution lemma*, stating that substituting a (closed,
171+--    well-typed) term `s` for a variable `x` in a term `t` preserves the
172+--    type of `t`. The proof goes by induction on the form of `t` and
173+--    requires looking at all the different cases in the definition of
174+--    substitution. This time, for the variables case, we discover that we
175+--    need to deduce from the fact that a term `s` has type S in the empty
176+--    context the fact that `s` has type S in every context. For this we
177+--    prove a...
178+
179+--  - *weakening* lemma, showing that typing is preserved under
180+--    "extensions" to the context `Γ`.
181+
182+--  To make Lean happy, though, we need to formalize the story in the
183+--  opposite order, starting with weakening...
184+
185+--  ### The Weakening Lemma
186+
187+--  First, we show that typing is preserved under "extensions" to the
188+--  context `Γ`. (Recall map inclusion, `Γ ⊆ Γ'`, from the `Typeclasses`
189+--  chapter.)
190+
191+-- IN PROGRESS
192+theorem weakening (Γ Γ' : Context) (t : Tm) (T : Ty)
193+    (hi : Γ ⊆ Γ') (hT : <{ ~Γ ⊢ ~t ⦂ ~T }>) : <{ ~Γ' ⊢ ~t ⦂ ~T }> := by
194+  induction hT generalizing Γ' with
195+  | var _ x _ h => exact .var _ x _ (hi h)
196+  | abs _ x _ _ _ _ ih => exact .abs _ x _ _ _ (ih _ (PartialMap.update_subset _ _ _ _ hi))
197+  | app _ _ _ _ _ _ _ ih₁ ih₂ => exact .app _ _ _ _ _ (ih₁ _ hi) (ih₂ _ hi)
198+  | tru => exact .tru _
199+  | fls => exact .fls _
200+  | ite _ _ _ _ _ _ _ _ ih₁ ih₂ ih₃ => exact .ite _ _ _ _ _ (ih₁ _ hi) (ih₂ _ hi) (ih₃ _ hi)
201+
202+--  The following simple corollary is what we actually need below.
203+
204+theorem weakening_empty (Γ : Context) (t : Tm) (T : Ty) (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) :
205+    <{ ~Γ ⊢ ~t ⦂ ~T }> :=
206+  weakening _ _ _ _
207+    (fun h => by
208+      rw [PartialMap.getElem_empty] at h
209+      cases h) hT
210+
211+--  ### The Substitution Lemma
212+
213+--  Now we come to the conceptual heart of the proof that reduction
214+--  preserves types — namely, the observation that *substitution* preserves
215+--  types.
216+
217+--  Formally, the so-called *substitution lemma* says this: Suppose we have
218+--  a term `t` with a free variable `x`, and suppose we've assigned a type
219+--  `T` to `t` under the assumption that `x` has some type `U`. Also,
220+--  suppose that we have some other term `v` and that we've shown that `v`
221+--  has type `U`. Then, since `v` satisfies the assumption we made about
222+--  `x` when typing `t`, we can substitute `v` for each of the occurrences
223+--  of `x` in `t` and obtain a new term that still has type `T`.
224+
225+-- IN PROGRESS
226+theorem substitution_preserves_typing (Γ : Context) (x : String) (U : Ty)
227+    (t v : Tm) (T : Ty)
228+    (hT : <{ ~x ↦ ~U ; ~Γ ⊢ ~t ⦂ ~T }>) (hv : <{ ∅ ⊢ ~v ⦂ ~U }>) :
229+    <{ ~Γ ⊢ [~x := ~v] ~t ⦂ ~T }> := by
230+  -- By induction on `t`; in each case we get at the derivation of `hT`.
231+  induction t generalizing Γ T with
232+  | var y =>
233+    cases hT with
234+    | var _ _ _ h =>
235+      by_cases hxy : x = y
236+      · subst hxy
237+        rw [PartialMap.update_eq] at h
238+        rw [subst_var_eq]
239+        have hUT : U = T := Option.some.inj h
240+        subst hUT
241+        exact weakening_empty _ _ _ hv
242+      · rw [PartialMap.update_neq hxy] at h
243+        rw [subst_var_ne _ _ _ hxy]
244+        exact .var _ y _ h
245+  | app t₁ t₂ ih₁ ih₂ =>
246+    cases hT with
247+    | app _ _ _ _ _ h₁ h₂ => rw [subst_app]; exact .app _ _ _ _ _ (ih₁ _ _ h₁) (ih₂ _ _ h₂)
248+  | abs y S t₁ ih =>
249+    cases hT with
250+    | abs _ _ _ _ _ h =>
251+      by_cases hxy : x = y
252+      · subst hxy
253+        rw [subst_abs_eq]
254+        rw [PartialMap.update_shadow] at h
255+        exact .abs _ _ _ _ _ h
256+      · rw [subst_abs_ne _ _ _ _ _ hxy]
257+        rw [PartialMap.update_permute (Ne.symm hxy)] at h
258+        exact .abs _ _ _ _ _ (ih _ _ h)
259+  | tru => cases hT with | tru => rw [subst_tru]; exact .tru _
260+  | fls => cases hT with | fls => rw [subst_fls]; exact .fls _
261+  | ite c t e ihc iht ihe =>
262+    cases hT with
263+    | ite _ _ _ _ _ h₁ h₂ h₃ =>
264+      rw [subst_ite]
265+      exact .ite _ _ _ _ _ (ihc _ _ h₁) (iht _ _ h₂) (ihe _ _ h₃)
266+
267+--  The substitution lemma can be viewed as a kind of "commutation
268+--  property." Intuitively, it says that substitution and typing can be
269+--  done in either order: we can either assign types to the terms `t` and
270+--  `v` separately (under suitable contexts) and then combine them using
271+--  substitution, or we can substitute first and then assign a type to
272+--  `[x:=v] t`; the result is the same either way.
273+
274+--  *Proof*: We show, by induction on `t`, that for all `T` and `Γ`, if
275+--  `x ↦ U; Γ ⊢ t ⦂ T` and `⊢ v ⦂ U`, then `Γ ⊢ [x:=v]t ⦂ T`.
276+
277+--  - If `t` is a variable there are two cases to consider, depending on
278+--    whether `t` is `x` or some other variable.
279+
280+--    - If `t = x`, then from the fact that `x ↦ U; Γ ⊢ x ⦂ T` we conclude
281+--      that `U = T`. We must show that `[x:=v]x = v` has type `T` under
282+--      `Γ`, given the assumption that `v` has type `U = T` under the empty
283+--      context. This follows from the weakening lemma.
284+
285+--    - If `t` is some variable `y` that is not equal to `x`, then we need
286+--      only note that `y` has the same type under `x ↦ U; Γ` as under `Γ`.
287+
288+--  - If `t` is an abstraction `λy:S. t₀`, then `T = S→T₁` and the IH tells
289+--    us, for all `Γ'` and `T₀`, that if `x ↦ U; Γ' ⊢ t₀ ⦂ T₀`, then
290+--    `Γ' ⊢ [x:=v]t₀ ⦂ T₀`. Moreover, by inspecting the typing rules we see
291+--    it must be the case that `y ↦ S; x ↦ U; Γ ⊢ t₀ ⦂ T₁`.
292+
293+--    The substitution in the conclusion behaves differently depending on
294+--    whether `x` and `y` are the same variable.
295+
296+--    First, suppose `x = y`. Then, by the definition of substitution,
297+--    `[x:=v]t = t`, so we just need to show `Γ ⊢ t ⦂ T`. Using
298+--    `HasType.abs`, we need to show that `y ↦ S; Γ ⊢ t₀ ⦂ T₁`. But we know
299+--    `y ↦ S; x ↦ U; Γ ⊢ t₀ ⦂ T₁`, and the claim follows since `x = y`.
300+
301+--    Second, suppose `x <> y`. Again, using `HasType.abs`, we need to show
302+--    that `y ↦ S; Γ ⊢ [x:=v]t₀ ⦂ T₁`. Since `x <> y`, we have
303+--    `y ↦ S; x ↦ U; Γ = x ↦ U; y ↦ S; Γ`. So we have
304+--    `x ↦ U; y ↦ S; Γ ⊢ t₀ ⦂ T₁`. Then, the the IH applies (taking
305+--    `Γ' = y ↦ S; Γ`), giving us `y ↦ S; Γ ⊢ [x:=v]t₀ ⦂ T₁`, as required.
306+
307+--  - If `t` is an application `t₁ t₂`, the result follows
308+--    straightforwardly from the definition of substitution and the
309+--    induction hypotheses.
310+
311+--  - The remaining cases are similar to the application case.
312+
313+--  One technical subtlety in the statement of the above lemma is that we
314+--  assume `v` has type `U` in the *empty* context — in other words, we
315+--  assume `v` is closed. (Since we are using a simple definition of
316+--  substitution that is not capture-avoiding, it doesn't make sense to
317+--  substitute non-closed terms into other terms. Fortunately, closed terms
318+--  are all we need!)
319+

332+--  ### Main Theorem
333+
334+--  We now have the ingredients we need to prove preservation: if a closed,
335+--  well-typed term `t` has type `T` and takes a step to `t'`, then `t'` is
336+--  also a closed term with type `T`. In other words, the small-step
337+--  reduction relation preserves types.
338+
339+-- IN PROGRESS
340+theorem preservation (t t' : Tm) (T : Ty)
341+    (hT : <{ ∅ ⊢ ~t ⦂ ~T }>) (hs : t ⟶ t') : <{ ∅ ⊢ ~t' ⦂ ~T }> := by
342+  generalize hΓ : (∅ : Context) = Γ at hT
343+  induction hT generalizing t' with
344+  | var => cases hs
345+  | abs => cases hs
346+  | tru => cases hs
347+  | fls => cases hs
348+  | app Γ T₁ T₂ t₁ t₂ h₁ h₂ ih₁ ih₂ =>
349+    subst hΓ
350+    cases hs with
351+    | appAbs _ _ _ _ _ =>
352+      -- The one interesting case: the desired result is the substitution lemma.
353+      cases h₁ with
354+      | abs _ _ _ _ _ hb => exact substitution_preserves_typing _ _ _ _ _ _ hb h₂
355+    | app1 _ t₁' _ h => exact .app _ _ _ _ _ (ih₁ t₁' h rfl) h₂
356+    | app2 _ _ t₂' _ h => exact .app _ _ _ _ _ h₁ (ih₂ t₂' h rfl)
357+  | ite Γ t₁ t₂ t₃ T₁ h₁ h₂ h₃ ih₁ ih₂ ih₃ =>
358+    subst hΓ
359+    cases hs with
360+    | ifTrue => exact h₂
361+    | ifFalse => exact h₃
362+    | ifStep _ t₁' _ _ h => exact .ite _ _ _ _ _ (ih₁ t₁' h rfl) h₂ h₃
363+
364+--  *Proof*: By induction on the derivation of `⊢ t ⦂ T`.
365+
366+--  - We can immediately rule out `HasType.var`, `HasType.abs`,
367+--    `HasType.tru`, and `HasType.fls` as final rules in the derivation,
368+--    since in each of these cases `t` cannot take a step.
369+
370+--  - If the last rule in the derivation is `HasType.app`, then
371+--    `t = t₁ t₂`, and there are subderivations showing that `⊢ t₁ ⦂ T₂→T`
372+--    and `⊢ t₂ ⦂ T₂` plus two induction hypotheses: (1) `t₁ ⟶ t₁'` implies
373+--    `⊢ t₁' ⦂ T₂→T` and (2) `t₂ ⟶ t₂'` implies `⊢ t₂' ⦂ T₂`. There are now
374+--    three subcases to consider, one for each rule that could be used to
375+--    show that `t₁ t₂` takes a step to `t'`.
376+
377+--    - If `t₁ t₂` takes a step by `Step.app1`, with `t₁` stepping to
378+--      `t₁'`, then, by the first IH, `t₁'` has the same type as `t₁`
379+--      (`⊢ t₁' ⦂ T₂→T`), and hence by `HasType.app` `t₁' t₂` has type `T`.
380+
381+--    - The `Step.app2` case is similar, using the second IH.
382+
383+--    - If `t₁ t₂` takes a step by `Step.appAbs`, then `t₁ = λx:T₀. t₀` and
384+--      `t₁ t₂` steps to `[x0:=t₂]t₀`; the desired result now follows from
385+--      the substitution lemma.
386+
387+--  - If the last rule in the derivation is `HasType.ite`, then
388+--    `t = if t₁ then t₂ else t₃`, with `⊢ t₁ ⦂ Bool`, `⊢ t₂ ⦂ T₁`, and
389+--    `⊢ t₃ ⦂ T₁`, and with three induction hypotheses: (1) `t₁ ⟶ t₁'`
390+--    implies `⊢ t₁' ⦂ Bool`, (2) `t₂ ⟶ t₂'` implies `⊢ t₂' ⦂ T₁`, and (3)
391+--    `t₃ ⟶ t₃'` implies `⊢ t₃' ⦂ T₁`.
392+
393+--    There are again three subcases to consider, depending on how `t`
394+--    steps.
395+
396+--    - If `t` steps to `t₂` or `t₃` by `Step.ifTrue` or `Step.ifFalse`,
397+--      the result is immediate, since `t₂` and `t₃` have the same type as
398+--      `t`.
399+
400+--    - Otherwise, `t` steps by `Step.ifStep`, and the desired conclusion
401+--      follows directly from the first induction hypothesis.
402+

443+--  ## Context Invariance (Optional)
444+
445+--  Another standard technical lemma associated with typed languages is
446+--  *context invariance*. It states that typing is preserved under
447+--  "inessential changes" to the context `Γ` — in particular, changes that
448+--  do not affect any of the free variables of the term. In this section,
449+--  we establish this property for our system, introducing some other
450+--  standard terminology on the way.
451+
452+--  First, we need to define the *free variables* in a term — i.e.,
453+--  variables that are used in the term in positions that are *not* in the
454+--  scope of an enclosing function abstraction binding a variable of the
455+--  same name.
456+
457+--  More technically, a variable `x` *appears free in* a term *t* if `t`
458+--  contains some occurrence of `x` that is not under an abstraction
459+--  labeled `x`. For example:
460+
461+--  - `y` appears free, but `x` does not, in `λx:T→U. x y`
462+--  - both `x` and `y` appear free in `(λx:T→U. x y) x`
463+--  - no variables appear free in `λx:T→U. λy:T. x y`
464+
465+--  We write this `x ∈ᶠ t`, reading the relation as "`x` is one of the free

558+--  we have a proof of some typing relation `Γ ⊢ t ⦂ T`, and we need to
559+--  replace `Γ` by a different context `Γ'`. When is it safe to do this?
560+--  Intuitively, it must at least be the case that `Γ'` assigns the same
561+--  types as `Γ` to all the variables that appear free in `t`. In fact,
562+--  this is the only condition that is needed.

566+    (hT : <{ ~Γ ⊢ ~t ⦂ ~T }>) (hf : ∀ x, x ∈ᶠ t → Γ[x] = Γ'[x]) :
567+    <{ ~Γ' ⊢ ~t ⦂ ~T }> := by

571+
572+--  - If the last rule in the derivation was `HasType.var`, then `t = x`
573+--    and `Γ x = T`. By assumption, `Γ' x = T` as well, and hence
574+--    `Γ' ⊢ t ⦂ T` by `HasType.var`.
575+
576+--  - If the last rule was `HasType.abs`, then `t = λy:T₂. t₁`, with
577+--    `T = T₂ → T₁` and `y ↦ T₂; Γ ⊢ t₁ ⦂ T₁`. The induction hypothesis
578+--    states that for any context `Γ''`, if `y ↦ T₂; Γ` and `Γ''` assign
579+--    the same types to all the free variables in `t₁`, then `t₁` has type
580+--    `T₁` under `Γ''`. Let `Γ'` be a context which agrees with `Γ` on the
581+--    free variables in `t`; we must show `Γ' ⊢ λy:T₂. t₁ ⦂ T₂ → T₁`.
582+
583+--    By `HasType.abs`, it suffices to show that `y ↦ T₂; Γ' ⊢ t₁ ⦂ T₁`. By
584+--    the IH (setting `Γ'' = y ↦ T₂;Γ'`), it suffices to show that
585+--    `y ↦ T₂;Γ` and `y ↦ T₂;Γ'` agree on all the variables that appear
586+--    free in `t₁`.
587+
588+--    Any variable occurring free in `t₁` must be either `y` or some other
589+--    variable. `y ↦ T₂; Γ` and `y ↦ T₂; Γ'` clearly agree on `y`.
590+--    Otherwise, note that any variable other than `y` that occurs free in
591+--    `t₁` also occurs free in `t = λy:T₂. t₁`, and by assumption `Γ` and
592+--    `Γ'` agree on all such variables; hence so do `y ↦ T₂; Γ` and
593+--    `y ↦ T₂; Γ'`.
594+
595+--  - If the last rule was `HasType.app`, then `t = t₁ t₂`, with
596+--    `Γ ⊢ t₁ ⦂ T₂ → T` and `Γ ⊢ t₂ ⦂ T₂`. One induction hypothesis states
597+--    that for all contexts `Γ'`, if `Γ'` agrees with `Γ` on the free
598+--    variables in `t₁`, then `t₁` has type `T₂ → T` under `Γ'`; there is a
599+--    similar IH for `t₂`. We must show that `t₁ t₂` also has type `T`
600+--    under `Γ'`, given the assumption that `Γ'` agrees with `Γ` on all the
601+--    free variables in `t₁ t₂`. By `HasType.app`, it suffices to show that
602+--    `t₁` and `t₂` each have the same type under `Γ'` as under `Γ`. But
603+--    all free variables in `t₁` are also free in `t₁ t₂`, and similarly
604+--    for `t₂`; hence the desired result follows from the induction
605+--    hypotheses.
